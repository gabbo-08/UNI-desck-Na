const CACHE_TTL_MS = 15 * 60 * 1000;
// Una sola acquisizione per studente: le tre rotte chiamate in parallelo dal
// frontend condividono la stessa promise invece di aprire tre sessioni Puppeteer.
const docentiCache = new Map(); // chiave profilo -> { promise, ts }

/**
 * Ricava email Docenti e password dalla sessione autenticata.
 * La password arriva solo dal JWT (req.user.password): mai da body o query string.
 */
function resolveCredentials(req) {
  const user = req.user || {};
  let userEmail = user.email;

  if (!userEmail || !userEmail.includes('@')) {
    const profile = getStudentProfile(db, user.sub);
    userEmail = profile?.email || userEmail;
  }
  // Se l'utente ha fatto login con l'email istituzionale, l'id profilo è l'email stessa.
  if ((!userEmail || !userEmail.includes('@')) && String(user.sub || '').includes('@')) {
    userEmail = String(user.sub).toLowerCase();
  }

  return { userEmail, password: user.password };
}

function isHardFailure(data) {
  return data && data.available === false && data.reason && data.reason !== 'DOCENTI_DATA_NOT_FOUND';
}

function getDocentiCached(profileKey, credentials) {
  const hit = docentiCache.get(profileKey);
  if (hit && Date.now() - hit.ts < CACHE_TTL_MS) return hit.promise;

  const promise = getDocentiData(credentials)
    .then(data => {
      // Non conservare in cache i fallimenti: il prossimo tentativo deve riprovare.
      if (isHardFailure(data)) docentiCache.delete(profileKey);
      return data;
    })
    .catch(error => {
      docentiCache.delete(profileKey);
      throw error;
    });

  docentiCache.set(profileKey, { promise, ts: Date.now() });
  return promise;
}

/**
 * Carica i dati Docenti per l'utente autenticato.
 * Restituisce i dati, oppure risponde direttamente con l'errore e restituisce null.
 */
async function loadForUser(req, res) {
  const { userEmail, password } = resolveCredentials(req);

  if (!userEmail || !userEmail.includes('@') || !password) {
    res.status(400).json({
      success: false,
      error: 'CREDENTIALS_REQUIRED',
      message: 'Email (@studenti.unina.it) o password non disponibili nella sessione. Esci e accedi di nuovo.'
    });
    return null;
  }

  try {
    const profileKey = String(req.user.sub);
    const data = await getDocentiCached(profileKey, { username: userEmail, password });

    if (isHardFailure(data)) {
      const status = data.reason === 'DOCENTI_EMAIL_REQUIRED' ? 400 : 502;
      const error = data.reason === 'DOCENTI_EMAIL_REQUIRED'
        ? 'CREDENTIALS_REQUIRED'
        : data.reason === 'DOCENTI_PRIVATE_LOGIN_FAILED'
          ? 'Login Docenti non riuscito: credenziali non valide.'
          : `Acquisizione Docenti non riuscita (${data.reason}).`;
      res.status(status).json({ success: false, error, reason: data.reason });
      return null;
    }

    return data;
  } catch (error) {
    res.status(500).json({ success: false, error: error.message || 'Errore Docenti.' });
    return null;
  }
}

// Serve al frontend dopo una ricarica pagina: al login i dati arrivano nella risposta di /api/auth/login,

const express = require('express');
const { getDatiCarriera } = require('../scraper/segrepass');
const careerCache = require('../services/career-cache');
const snapshots = require('../services/career-snapshots');
const { sessionErrorMessage } = require('../lib/errors');

const router = express.Router();

router.get('/', async (req, res) => {
  const profileId = req.user.sub;
  const codiceFiscale = req.user.codiceFiscale || profileId;
  const { password } = req.user;

  // Token vecchio, senza password cifrata: serve un nuovo login.
  // 401 => il frontend mostra "Sessione scaduta" e rimanda al form di accesso.
  if (!password) {
    return res.status(401).json({ error: 'Sessione non più valida: accedi di nuovo.' });
  }

  try {
    const { value } = await careerCache.getOrLoadCareer(
      profileId,
      password,
      () => getDatiCarriera({ codiceFiscale, password }, { phase: 'fast' })
    );

    // Se i dati lenti (tasse, appelli) sono già pronti li uniamo subito.
    const slow = snapshots.getSlowSnapshot(profileId);
    return res.json({
      ...value,
      tasse: slow?.taxes ?? value?.tasse ?? null,
      prenotazioni: slow?.prenotazioni ?? value?.prenotazioni ?? []
    });
  } catch (error) {
    console.error('[Metriche Error]', error.message);
    // 502 (non 401/403): un errore del portale non deve azzerare la sessione.
    return res.status(502).json({ error: sessionErrorMessage(error) });
  }
});

function isConfigured(profile) {
  return Boolean(profile?.course_id && profile?.course_name && profile?.cohort_year && profile?.current_course_year && profile?.channel);
}

function parseProfileUpdate(body) {
  const updates = {};
  const stringFields = ['courseId', 'courseName', 'channel', 'surnameInitial'];
  for (const field of stringFields) {
    if (!Object.hasOwn(body, field)) continue;
    if (body[field] !== null && (typeof body[field] !== 'string' || body[field].length > 120)) {
      return { error: `${field} deve essere una stringa di massimo 120 caratteri o null.` };
    }
    updates[field] = typeof body[field] === 'string' ? body[field].trim() || null : null;
  }

  for (const [field, min, max] of [['cohortYear', 1900, 2200], ['currentCourseYear', 1, 20]]) {
    if (!Object.hasOwn(body, field)) continue;
    if (body[field] !== null && (!Number.isInteger(Number(body[field])) || Number(body[field]) < min || Number(body[field]) > max)) {
      return { error: `${field} non valido.` };
    }
    updates[field] = body[field] === null ? null : Number(body[field]);
  }

  if (Object.hasOwn(body, 'selectedElectiveIds') &&
    (!Array.isArray(body.selectedElectiveIds) || body.selectedElectiveIds.length > 100 ||
      body.selectedElectiveIds.some(id => !Number.isInteger(Number(id)) || Number(id) < 1))) {
    return { error: 'selectedElectiveIds deve essere un array di identificativi positivi.' };
  }

  return { updates, selectedElectiveIds: body.selectedElectiveIds };
}

router.get('/', (req, res) => {
  const profile = getStudentProfile(db, req.user.sub);
  if (!profile) return res.status(404).json({ error: 'Profilo non trovato.' });
  return res.json({
    profile,
    selectedElectiveIds: getSelectedElectiveIds(db, req.user.sub),
    onboardingRequired: !isConfigured(profile)
  });
});

router.put('/', (req, res, next) => {
  try {
    const parsed = parseProfileUpdate(req.body || {});
    if (parsed.error) return res.status(400).json({ error: parsed.error });

    const current = getStudentProfile(db, req.user.sub);
    if (!current) return res.status(404).json({ error: 'Profilo non trovato.' });
    const courseChanged = (Object.hasOwn(parsed.updates, 'courseId') && parsed.updates.courseId !== current.course_id) ||
      (Object.hasOwn(parsed.updates, 'cohortYear') && parsed.updates.cohortYear !== current.cohort_year);

    const profile = updateStudentProfile(db, req.user.sub, parsed.updates);
    if (courseChanged && !parsed.selectedElectiveIds) clearSelectedElectives(db, req.user.sub);

    if (parsed.selectedElectiveIds) {
      const plan = getStudyPlan(db, profile.course_id, profile.cohort_year);
      if (!plan) return res.status(409).json({ error: 'Carica prima il piano di studi per salvare gli esami a scelta.' });
      replaceSelectedElectives(db, req.user.sub, plan.id, parsed.selectedElectiveIds.map(Number));
    }

    return res.json({
      profile,
      selectedElectiveIds: getSelectedElectiveIds(db, req.user.sub),
      onboardingRequired: !isConfigured(profile)
    });
  } catch (error) {
    return next(error);
  }
});

const express = require('express');
const { db } = require('../db');
const { getStudentProfile } = require('../db/repository');
const { getTimetable } = require('../services/easyacademy-provider');

const router = express.Router();

router.get('/', async (req, res, next) => {
  const profile = getStudentProfile(db, req.user.sub);
  if (!profile) return res.status(404).json({ error: 'Profilo non trovato.' });

  const studyYear = req.query.studyYear === undefined ? profile.current_course_year : Number(req.query.studyYear);
  const semester = req.query.semester === undefined ? null : Number(req.query.semester);
  const channel = req.query.channel === undefined ? profile.channel : String(req.query.channel).trim();
  const academicYear = req.query.academicYear === undefined ? null : String(req.query.academicYear).trim();

  if (!Number.isInteger(studyYear) || studyYear < 1 || studyYear > 20) {
    return res.status(400).json({ error: 'studyYear non valido.' });
  }
  if (semester !== null && ![1, 2].includes(semester)) {
    return res.status(400).json({ error: 'semester deve essere 1 o 2.' });
  }
  if (!channel) return res.status(409).json({ code: 'PROFILE_ONBOARDING_REQUIRED', error: 'Imposta il canale nel profilo.' });
  if (academicYear && !/^\d{4}(?:\/\d{2,4})?$/.test(academicYear)) {
    return res.status(400).json({ error: 'academicYear non valido.' });
  }

  try {
    const timetable = await getTimetable({
      courseId: profile.course_id,
      courseName: profile.course_name,
      academicYear,
      studyYear,
      semester,
      channel,
      surnameInitial: profile.surname_initial
    });
    return res.json({ filters: { academicYear, studyYear, semester, channel }, timetable });
  } catch (error) {
    if (error.status) return res.status(error.status).json({ code: error.code, error: error.message });
    return next(error);
  }
});

function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

function normalizeLabel(value) {
  return String(value || '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .trim();
}

function toArray(json) {
  if (Array.isArray(json)) return json;
  if (Array.isArray(json?.content)) return json.content;
  if (Array.isArray(json?.data)) return json.data;
  if (Array.isArray(json?.items)) return json.items;
  return [];
}

function pick(item, keys, fallback = '') {
  for (const key of keys) {
    const value = item?.[key];
    if (value !== undefined && value !== null && String(value).trim() !== '') {
      return value;
    }
  }
  return fallback;
}

function professorName(item) {
  const nome = pick(item, ['nomeDocente', 'nome_docente', 'nome']);
  const cognome = pick(item, ['cognomeDocente', 'cognome_docente', 'cognome']);
  const full = `${nome} ${cognome}`.trim();
  if (full && full !== PLACEHOLDER_PROFESSOR) return full;

  const d = item?.docente ?? item?.professore ?? item?.titolare;
  if (d && typeof d === 'object') {
    const fullObj = `${d.nome || ''} ${d.cognome || ''}`.trim();
    if (fullObj) return fullObj;
  }
  if (typeof d === 'string' && d.trim()) return d.trim();

  return PLACEHOLDER_PROFESSOR;
}

function courseNameOf(item) {
  const raw = pick(item, [
    'descrInse', 'nomeInse', 'nomeInsegnamento', 'insegnamento', 'denominazioneInsegnamento',
    'denominazione', 'materia', 'nome', 'descrizione'
  ], PLACEHOLDER_COURSE);
  if (raw && typeof raw === 'object') {
    return pick(raw, ['nome', 'denominazione', 'descrizione'], PLACEHOLDER_COURSE);
  }
  return String(raw);
}

// Le date del portale arrivano come millisecondi (es. 1793228400000): le converto
// in formato italiano leggibile (dd/mm/yyyy hh:mm, ora di Roma).
function formatDocenteDate(value) {
  if (value === undefined || value === null || String(value).trim() === '') return '';
  const text = String(value).trim();
  if (/^\d{12,14}$/.test(text)) {
    const date = new Date(Number(text));
    if (!Number.isNaN(date.getTime())) {
      const giorno = date.toLocaleDateString('it-IT', { timeZone: 'Europe/Rome', day: '2-digit', month: '2-digit', year: 'numeric' });
      const ora = date.toLocaleTimeString('it-IT', { timeZone: 'Europe/Rome', hour: '2-digit', minute: '2-digit' });
      return ora === '00:00' ? giorno : `${giorno} ${ora}`;
    }
  }
  return text;
}

function dedupe(list, keyFn) {
  const seen = new Set();
  return list.filter(item => {
    const key = keyFn(item);
    if (seen.has(key)) return false;
    seen.add(key);
    return true;
  });
}

function teacherBoardUrl(teacherId) {
  return `${DOCENTI_ORIGIN}/teacher/${encodeURIComponent(teacherId)}/dashboard/advices`;
}

async function fetchDocentiJson(path) {
  try {
    const response = await fetch(`${DOCENTI_API}${path}`, {
      headers: {
        Accept: 'application/json',
        'Accept-Language': 'it-IT,it;q=0.9',
        Origin: DOCENTI_ORIGIN,
        Referer: `${DOCENTI_ORIGIN}/`,
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)'
      },
      signal: AbortSignal.timeout(HTTP_TIMEOUT_MS)
    });
    if (!response.ok) return null;
    const contentType = response.headers.get('content-type') || '';
    if (!contentType.includes('json')) return null;
    return await response.json();
  } catch (error) {
    return null;
  }
}

async function findTeacher(fullName, dependencies = {}) {
  const query = String(fullName || '').trim();
  if (query.length < 3) return null;

  const fetchJson = dependencies.fetchJson || fetchDocentiJson;
  const tokens = normalizeLabel(query).split(/\s+/).filter(t => t.length > 2);
  const attempts = [query, ...[...tokens].reverse()];

  let candidates = [];
  for (const attempt of [...new Set(attempts)]) {
    const data = await fetchJson(`/docenti?nome=${encodeURIComponent(attempt)}&p=0&s=10`);
    candidates = Array.isArray(data?.content) ? data.content : [];
    if (candidates.length > 0) break;
  }
  if (candidates.length === 0) return null;

  const scored = candidates
    .map(candidate => {
      const label = normalizeLabel(`${candidate.nome || ''} ${candidate.cognome || ''}`);
      const score = tokens.filter(t => label.includes(t)).length;
      return { candidate, score };
    })
    .sort((a, b) => b.score - a.score);

  const best = scored[0];
  if (!best || best.score === 0) return null;

  const tied = scored.filter(s => s.score === best.score);
  if (tied.length > 1) return null;

  return best.candidate;
}

function stripHtml(html) {
  return String(html || '')
    .replace(/<\s*br\s*\/?>/gi, '\n')
    .replace(/<\/(p|div|li|h[1-6]|tr)>/gi, '\n')
    .replace(/<li[^>]*>/gi, '• ')
    .replace(/<[^>]+>/g, '')
    .replace(/&nbsp;/g, ' ')
    .replace(/&lt;/g, '<')
    .replace(/&gt;/g, '>')
    .replace(/&quot;/g, '"')
    .replace(/&#39;/g, "'")
    .replace(/&amp;/g, '&')
    .replace(/[ \t]+\n/g, '\n')
    .replace(/\n{3,}/g, '\n\n')
    .trim();
}

function mapAnnouncement(item, teacherId) {
  const rawAttachments = Array.isArray(item?.allegati) ? item.allegati : [];
  const attachments = rawAttachments
    .filter(a => a && a.id)
    .map(a => ({
      id: String(a.id),
      name: String(pick(a, ['nomeFile', 'nome', 'filename', 'titolo', 'descrizione'], `Allegato ${a.id}`)),
      url: `${DOCENTI_API}/allegati/avvisi/${a.id}`
    }));

  let category = pick(item, ['categoria', 'tipologia', 'tipo'], null);
  if (category && typeof category === 'object') {
    category = pick(category, ['descrizione', 'nome', 'label'], null);
  }

  const content = String(pick(item, ['testo', 'descrizione', 'contenuto', 'messaggio', 'corpo'], ''));

  return {
    id: String(item?.id ?? ''),
    title: String(pick(item, ['titolo', 'oggetto', 'title'], 'Avviso')),
    content,
    contentText: stripHtml(content),
    date: pick(item, ['dataPubb', 'dataPubblicazione', 'dataInserimento', 'dataAgg'], null),
    updatedAt: pick(item, ['dataAgg', 'dataAggiornamento', 'dataModifica'], null),
    category: category ? String(category) : null,
    attachments,
    link: attachments[0]?.url || null,
    boardUrl: teacherId ? teacherBoardUrl(teacherId) : null
  };
}

async function fetchAnnouncementDetail(teacherId, announcementId, dependencies = {}) {
  const fetchJson = dependencies.fetchJson || fetchDocentiJson;
  const tid = encodeURIComponent(teacherId);
  const aid = encodeURIComponent(announcementId);
  const candidates = [
    `/docenti/${tid}/avvisi/${aid}`,
    `/avvisi/${aid}`,
    `/docenti/avvisi/${aid}`
  ];

  for (const path of candidates) {
    const data = await fetchJson(path);
    const isSingleObject = data && typeof data === 'object' && !Array.isArray(data) && !Array.isArray(data.content);
    if (isSingleObject) return data;
  }
  return null;
}

function timestampOf(value) {
  const t = Date.parse(value);
  return Number.isNaN(t) ? 0 : t;
}

async function fetchAnnouncementsForProfessors(professors = [], dependencies = {}) {
  const fetchJson = dependencies.fetchJson || fetchDocentiJson;
  const names = [...new Set(
    professors
      .map(p => String(p || '').trim())
      .filter(p => p && !/non specificato/i.test(p))
  )].slice(0, MAX_ANNOUNCEMENT_TEACHERS);

  const groups = await Promise.all(names.map(async professor => {
    const teacher = await findTeacher(professor, { fetchJson });
    if (!teacher?.id) return null;

    const data = await fetchJson(
      `/docenti/${encodeURIComponent(teacher.id)}/avvisi?p=0&s=${ANNOUNCEMENTS_PER_TEACHER}`
    );
    const raw = toArray(data);
    if (raw.length === 0) return null;

    const items = await Promise.all(raw.map(async rawItem => {
      let mapped = mapAnnouncement(rawItem, teacher.id);
      if (!mapped.content && mapped.id) {
        const detail = await fetchAnnouncementDetail(teacher.id, mapped.id, { fetchJson });
        if (detail) mapped = mapAnnouncement({ ...rawItem, ...detail }, teacher.id);
      }
      return mapped;
    }));
    items.sort((a, b) => timestampOf(b.date) - timestampOf(a.date));

    return {
      professor: `${teacher.nome || ''} ${teacher.cognome || ''}`.trim() || professor,
      slug: String(teacher.id),
      boardUrl: teacherBoardUrl(teacher.id),
      items
    };
  }));

  return groups.filter(Boolean);
}

function startApiCollector(page) {
  const captured = [];
  const pending = [];

  const onResponse = response => {
    const url = response.url();
    if (!url.includes('/webdocenti-be/')) return;
    if (response.request().method() !== 'GET' || !response.ok()) return;
    if (!(response.headers()['content-type'] || '').includes('json')) return;

    pending.push(
      response.json()
        .then(json => { captured.push({ url, json }); })
        .catch(() => {})
    );
  };

  page.on('response', onResponse);

  return {
    async settle() {
      await Promise.all(pending);
      return captured;
    },
    find(pattern) {
      const hit = captured.find(c => pattern.test(c.url));
      return hit ? hit.json : null;
    },
    findAll(pattern) {
      return captured.filter(c => pattern.test(c.url)).map(c => c.json);
    },
    stop() {
      page.off('response', onResponse);
    }
  };
}

      );

        subscriptions,
        bookedExams

    });
  } catch (err) {
    console.warn('[Docenti] Errore scraping:', err.message || err);

  }
}

 * Esempio: "Insegnamenti del Piano: PIANO GENERALE (IMM. DAL 2025/26) (DE301)"

function parsePianoHeader(text = '') {
  const source = String(text || '');
  const cohort = source.match(/IMM\.?\s*DAL\s*(\d{4})/i);
  const code = source.match(/Insegnamenti del Piano:[^\n]*?\(([A-Z0-9]{3})[A-Z0-9]*\)/i);

  return {
    annoImmatricolazione: cohort ? Number(cohort[1]) : null,
    codiceCorso: code ? code[1] : null
  };
}

async function openPianoPage(page, esisUrl) {
  const origin = new URL(esisUrl).origin;
  // ESIS può completare il rendering dopo il timeout: il contenuto resta leggibile.
  await page.goto(`${origin}${PIANO_PATH}`, { waitUntil: 'domcontentloaded', timeout: 30000 })
    .catch(error => console.warn('[piano] Navigazione lenta:', error.message));

  await page.waitForFunction(
    () => /Insegnamenti del Piano|Piano di Studi/i.test(document.body ? document.body.innerText : ''),
    { timeout: 6000 }
  ).catch(() => {});
  await new Promise(resolve => setTimeout(resolve, 1200));
}

/**
 * Trova la tabella degli insegnamenti: è quella la cui intestazione contiene
 * Codice, Descrizione e CFU. Restituisce intestazioni e righe come testo.
 */
async function extractPianoTable(page) {
  return page.evaluate(() => {
    const clean = value => (value || '').replace(/\s+/g, ' ').trim();

    for (const table of Array.from(document.querySelectorAll('table'))) {
      const rows = Array.from(table.querySelectorAll('tr'));
      const headerIndex = rows.findIndex(row => {
        const text = clean(row.innerText);
        return /\bCodice\b/i.test(text) && /\bCFU\b/i.test(text) && /Descrizione/i.test(text);
      });
      if (headerIndex === -1) continue;

      const headers = Array.from(rows[headerIndex].children).map(cell => clean(cell.innerText));
      const body = rows
        .slice(headerIndex + 1)
        .map(row => Array.from(row.children).map(cell => clean(cell.innerText)))
        .filter(cells => cells.some(Boolean));

      return { headers, rows: body };
    }
    return null;
  });
}

    puppeteerOptions: {
      headless: 'new',
      ignoreHTTPSErrors: true,
      args: [
        '--no-sandbox',
        '--disable-setuid-sandbox',
        '--disable-dev-shm-usage',

      ]
    },
    // Il login Segrepass può richiedere due tentativi di nodo + estrazione
    // esami, tasse e appelli: 120s erano insufficienti e chiudevano sessioni valide.
    timeout: 240000,
  });

  cluster.on('taskerror', (err, data) => {
    console.error(`[Pool Error] Fallimento task di scraping: ${err.message}`);
  });

  console.log('[Pool] Puppeteer Cluster operativo (Max Concurrency: 3)');
  return cluster;
}

async function queueScrapingTask(data, scrapeLogic) {
  if (!cluster) throw new Error("Cluster non inizializzato.");
  return await cluster.execute(data, async ({ page, data }) => {
    return await scrapeLogic(page, data);
  });
}

  if (cluster) {
    console.log('[Pool] Chiusura cluster in corso...');
    await cluster.idle();
    await cluster.close();
    cluster = null;
    console.log('[Pool] Cluster disattivato.');
  }
}

        const cnpMatch = text.match(/Codice\s+Numerico\s+Personale[^\d]*(\d+)/i);

        if (pinMatch && !pin) pin = pinMatch[1].trim();
        if (cnpMatch && !cnp) cnp = cnpMatch[1].trim();
      } catch {}
    }

    return { pin, codiceNumericoPersonale: cnp };
  } catch (err) {
    debugLog('[Segrepass] Errore durante l\'estrazione del PIN:', err.message);
    return { pin: null, codiceNumericoPersonale: null };
  } finally {
    if (page) await page.close().catch(() => {});
  }
}

// utils.js
async function enableFastMode(page) {
  try {
    await page.setRequestInterception(true);
    page.on('request', (req) => {
      if (req.isInterceptResolutionHandled()) return;
      // Blocca risorse inutili per velocizzare il caricamento
      if (['image', 'font', 'media', 'stylesheet'].includes(req.resourceType())) {
        req.abort().catch(() => {});
      } else {
        req.continue().catch(() => {});
      }
    });
  } catch (err) {
    console.error('[Utils] Errore in enableFastMode:', err.message);
  }
}

async function extractTables(targetPage) {
  const results = [];
  for (const frame of targetPage.frames()) {
    try {
      const tables = await frame.evaluate(() => {
        const cleanStr = s => (s || '').replace(/<[^>]*>/g, '').replace(/\s+/g, ' ').trim();
        return Array.from(document.querySelectorAll('table')).map(t => {
          const rows = Array.from(t.querySelectorAll('tr')).map(tr => ({
            cells: Array.from(tr.querySelectorAll('th, td')).map(td => cleanStr(td.innerText)),
          }));
          
          if (rows.length < 2) return null;
          
          const headers = rows[0].cells;
          const records = rows.slice(1)
            .map(row => row.cells)
            .filter(row => row.some(cell => cell !== ''))
            .map(row => Object.fromEntries(headers.map((h, i) => [h || `col_${i}`, row[i] ?? ''])));
            
          return { headers, records, rowCount: records.length };
        }).filter(Boolean);
      });
      if (tables && tables.length > 0) results.push(...tables);
    } catch {}
  }
  return results;
}

async function call(path, { method = 'GET', body, cookie, timeout = 30000 } = {}) {
  const headers = { Origin: ORIGIN, Accept: 'application/json' };
  if (body !== undefined) headers['Content-Type'] = 'application/json';
  if (token) headers.Authorization = `Bearer ${token}`;
  if (cookie) headers.Cookie = cookie;

  return fetch(`${BASE_URL}${path}`, {
    method,
    headers,
    body: body === undefined ? undefined : JSON.stringify(body),
    signal: AbortSignal.timeout(timeout)
  });
}

const crypto = require('node:crypto');
const { JWT_SECRET } = require('../lib/config');

const DEFAULT_TTL_MS = 15 * 60 * 1000;
const cache = new Map();
const inFlight = new Map();

function ttlMs() {
  const configured = Number(process.env.CAREER_CACHE_TTL_MS);
  return Number.isFinite(configured) && configured >= 0 ? configured : DEFAULT_TTL_MS;
}

/**
 * Chiave per studente + credenziale. Della password resta solo un HMAC:
 * una password errata non può quindi riutilizzare i dati in cache.
 */
function cacheKey(profileId, password) {
  const fingerprint = crypto
    .createHmac('sha256', JWT_SECRET)
    .update(String(password || ''))
    .digest('hex');
  return `${profileId}:${fingerprint}`;
}

function prune(now = Date.now()) {
  const ttl = ttlMs();
  for (const [key, entry] of cache) {
    if (now - entry.storedAt > ttl) cache.delete(key);
  }
}

function getCachedCareer(profileId, password, now = Date.now()) {
  const ttl = ttlMs();
  if (ttl <= 0) return null;

  const key = cacheKey(profileId, password);
  const entry = cache.get(key);
  if (!entry) return null;
  if (now - entry.storedAt > ttl) {
    cache.delete(key);
    return null;
  }
  return { value: entry.value, storedAt: entry.storedAt };
}

function setCachedCareer(profileId, password, value, now = Date.now()) {
  if (ttlMs() <= 0) return;

// Trasforma le righe della tabella "Esami Sostenuti" di Segrepass in oggetti Esame
// (stesso formato di api.ts) e calcola CFU, medie e base di laurea.

const normalizeKey = value => String(value ?? '')
  .normalize('NFD')
  .replace(/[\u0300-\u036f]/g, '')
  .toLowerCase()
  .trim();

const cleanText = value => String(value ?? '').replace(/\s+/g, ' ').trim();

/**
 * Primo valore non vuoto la cui intestazione corrisponde a uno dei pattern (in ordine di priorita')
 */
function pickField(record, patterns, exclude = null) {
  const entries = Object.entries(record);
  for (const pattern of patterns) {
    for (const [key, value] of entries) {
      const normalized = normalizeKey(key);
      if (exclude && exclude.test(normalized)) continue;
      if (pattern.test(normalized) && cleanText(value)) return cleanText(value);
    }
  }
  return null;
}

/**
 * "28", "30 e lode", "30L", "28/30" -> voto; "IDONEO" -> giudizio
 */
function parseGrade(raw) {
  const text = cleanText(raw).toUpperCase();
  if (!text) return { voto: null, lode: false, giudizio: null };
  const digits = text.match(/(\d{2})/);
  const voto = digits ? Number(digits[1]) : null;
  if (voto !== null && voto >= 18 && voto <= 30) {
    return { voto, lode: voto === 30 && /LODE|30\s*L\b/.test(text), giudizio: null };
  }
  return { voto: null, lode: false, giudizio: cleanText(raw) };
}

const PASSED_JUDGMENT = /idone|approvat|superat|positiv|conseguit|riconosciut|convalid/i;

/**
 * Righe del libretto -> Esame[] (solo esami superati)
 */
function normalizeEsami(records = []) {
  const hasGradeColumn = records.some(record =>
    Object.keys(record).some(key => /voto|valutazione|esito|giudizio/.test(normalizeKey(key)))
  );

  const exams = [];
  for (const record of records) {
    const name = pickField(
      record,
      [/insegnamento/, /denominazione/, /attivita/, /descrizione/, /materia/, /^esame/],
      /codice|^cod$/
    );
    if (!name) continue;

    const gradeRaw = pickField(record, [/^voto/, /valutazione/, /esito/, /giudizio/]);
    const { voto, lode, giudizio } = parseGrade(gradeRaw);
    // Senza colonna dei voti nella tabella, tutte le righe sono considerate superate
    const passed = !hasGradeColumn || voto !== null || (giudizio && PASSED_JUDGMENT.test(giudizio));
    if (!passed) continue;

    const cfuRaw = pickField(record, [/cfu/, /crediti/, /peso/]);
    const cfu = cfuRaw ? Number((cfuRaw.replace(',', '.').match(/\d+(?:\.\d+)?/) || ['0'])[0]) : 0;

    exams.push({
      materia: name,
      insegnamento: name,
      cfu: Number.isFinite(cfu) ? cfu : 0,
      voto,
      giudizio: voto === null ? giudizio : null,
      lode,
      data: pickField(record, [/data/, /verbalizz/, /superamento/]) || undefined
    });
  }
  return exams;
}

/**
 * Metriche nel formato atteso dal frontend (tipo Metriche di api.ts)
 */
function buildMetriche(records = [], { cfuTotali = 180 } = {}) {
  const esami = normalizeEsami(records);
  const graded = esami.filter(exam => exam.voto !== null);

  const cfuAcquisiti = esami.reduce((sum, exam) => sum + exam.cfu, 0);
  const gradedCfu = graded.reduce((sum, exam) => sum + exam.cfu, 0);

  const mediaAritmetica = graded.length
    ? graded.reduce((sum, exam) => sum + exam.voto, 0) / graded.length
    : null;
  const mediaPonderata = gradedCfu > 0
    ? graded.reduce((sum, exam) => sum + exam.voto * exam.cfu, 0) / gradedCfu
    : null;

  const format = value => (value === null ? null : value.toFixed(2));
  return {
    esami,
    cfuAcquisiti,
    cfuTotali,
    avanzamentoCFU: Math.min(100, Math.round((cfuAcquisiti / cfuTotali) * 100)),
    mediaAritmetica: format(mediaAritmetica),
    mediaPonderata: format(mediaPonderata),
    baseLaurea: mediaPonderata === null ? null : format((mediaPonderata * 110) / 30)
  };
}

module.exports = { buildMetriche, normalizeEsami, parseGrade };

/**

function resolveCourse(courseCode, fallbackName = '') {
  if (!courseCode) {
    return getFallbackCourse(fallbackName);
  }

  const cleanCode = String(courseCode).trim().toUpperCase();

  if (coursesMap[cleanCode]) {
    return {
      code: cleanCode,

    };

  return getFallbackCourse(fallbackName, cleanCode);
}

  const cleanName = String(fallbackName).toUpperCase();

      return { code: key, ...course };
    }
  }

  return {
    code,
    name: fallbackName || 'CORSO GENERICO',
    department: 'DIPARTIMENTO GENERALE',
    areaId: 'generale',
    easyAcademyId: code,
    degreeType: 'L'
  };
}

// Cache breve per i dati del portale Docenti: evita di riaprire un browser

const crypto = require('node:crypto');
const { getDocentiSnapshot, isDocentiLoading } = require('./career-snapshots');
const { JWT_SECRET } = require('../lib/config');

const TTL_MS = 5 * 60 * 1000;
// Il login avvia l'acquisizione in background: le richieste che arrivano nel
// frattempo attendono lo stesso job invece di avviare altri browser.
const LOADING_WAIT_MS = 60000;
const LOADING_POLL_MS = 500;
const cache = new Map();
const inFlight = new Map();

function delay(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

function credentialFingerprint(credentials) {
  if (!credentials) return '';
  const username = credentials.username || credentials.codiceFiscale || '';
  const password = credentials.password || '';
  if (!username && !password) return '';
  return crypto
    .createHmac('sha256', JWT_SECRET)
    .update(`${username}\u0000${password}`)
    .digest('hex')
    .slice(0, 16);
}

function cacheKey(profileId, credentials) {
  const fingerprint = credentialFingerprint(credentials);
  return fingerprint ? `${profileId}:${fingerprint}` : String(profileId);
}

async function waitForBackground(profileId) {
  const deadline = Date.now() + LOADING_WAIT_MS;
  while (Date.now() < deadline) {
    const snapshot = getDocentiSnapshot(profileId);
    if (snapshot) return snapshot;
    if (!isDocentiLoading(profileId)) return null;
    await delay(LOADING_POLL_MS);
  }
  return getDocentiSnapshot(profileId);
}

function normalizeArgs(profileId, second, third, fourth) {
  if (typeof second === 'function') {
    return { credentials: fourth || third || null, loader: second };
  }
  return { credentials: second || null, loader: third };
}

async function getDocentiCached(profileId, second, third, fourth) {
  const { credentials, loader } = normalizeArgs(profileId, second, third, fourth);
  const key = cacheKey(profileId, credentials);

  const entry = cache.get(key);
  if (entry && Date.now() - entry.storedAt < TTL_MS) return entry.value;

  // Il login carica già i dati in background: se ci sono, usali subito.
  const background = getDocentiSnapshot(profileId);
  if (background) {
    cache.set(key, { value: background, storedAt: Date.now() });
    return background;
  }

  if (isDocentiLoading(profileId)) {
    const loaded = await waitForBackground(profileId);
    if (loaded) {
      cache.set(key, { value: loaded, storedAt: Date.now() });
      return loaded;
    }
  }

  if (inFlight.has(key)) return inFlight.get(key);

  const task = Promise.resolve()
    .then(loader)
    .then(value => {
      cache.set(key, { value, storedAt: Date.now() });
      return value;
    })
    .finally(() => inFlight.delete(key));

  inFlight.set(key, task);
  return task;
}

function clearDocentiCache(profileId) {
  if (profileId) {
    const prefix = String(profileId);
    for (const key of [...cache.keys()]) {
      if (key === prefix || key.startsWith(`${prefix}:`)) cache.delete(key);
    }
    for (const key of [...inFlight.keys()]) {
      if (key === prefix || key.startsWith(`${prefix}:`)) inFlight.delete(key);
    }
  } else {
    cache.clear();
    inFlight.clear();
  }
}

function getCourseOptions(options, { courseId, courseName, studyYear }) {
  const wantedName = normalizeCourseName(String(courseName || '')
    .replace(/\[[^\]]+\]/g, '')
    .replace(/\bL-?\s*8\b/i, ''));
  const candidates = options.map(option => {
    const text = String(option.text || '').trim();
    const isThirdYear = /terzo anno/i.test(text);
    const displayName = normalizeCourseName(text.replace(/\s*\([^)]*\)/g, '').replace(/\s*[-–]\s*terzo anno/i, ''));
    const codeMatch = Boolean(courseId && String(option.value).toLowerCase() === String(courseId).toLowerCase());
    const similarity = wantedName ? nameSimilarity(displayName, wantedName) : 0;
    const yearMatch = (studyYear >= 3) === isThirdYear;
    return { option, text, codeMatch, similarity, score: similarity + (yearMatch ? 0.25 : 0) + (codeMatch ? 0.1 : 0) };
  }).filter(candidate => candidate.option.value && (candidate.codeMatch || candidate.similarity >= 0.72));

  candidates.sort((left, right) => right.score - left.score);

function extractGroupRange(text) {
  const token = String(text || '').toUpperCase().match(/\b([A-Z])\s*[-–]\s*([A-Z]+)/);
  if (!token) return null;

function matchesStudyGroup(optionText, channel, surnameInitial) {
  const normalizedOption = normalizeLabel(optionText);
  const channelToken = normalizeLabel(String(channel || '').split(':').pop().replace(/corso generico/i, ''));
  if (channelToken && normalizedOption.includes(channelToken)) return true;

  const channelPart = String(surnameInitial || channel || '').toUpperCase().split(':').pop();
  const initial = channelPart.match(/[A-Z]/)?.[0];
  const range = extractGroupRange(optionText);
  return Boolean(initial && range && initial >= range.first && initial <= range.last);
}

// Solo DE3 e' verificato (compare cosi' sul portale Docenti: "DE3 INGEGNERIA INFORMATICA").
// Per gli altri corsi si usa il codice letto da Segrepass oppure il nome normalizzato:
// i codici DE1, DE2, DE4 e DE5 della versione precedente erano inventati e sono stati tolti.
const COURSE_CODE_MAP = {
  'INGEGNERIA INFORMATICA': 'DE3'
};

// ATTENZIONE: solo "A-D" e' confermato dal tuo profilo. Gli altri intervalli sono un'ipotesi:
// confrontali con i canali che mostra EasyAcademy per il tuo corso.
const DEFAULT_CANALI = [
  { da: 'A', a: 'D', canale: 'A-D' },
  { da: 'E', a: 'O', canale: 'E-O' },
  { da: 'P', a: 'Z', canale: 'P-Z' }
];

const MAX_COURSE_YEARS = 3;

/**
 * Anno di inizio dell'anno accademico in corso (l'anno accademico parte a settembre)
 */
function currentAcademicStart(now = new Date()) {
  return now.getMonth() >= 8 ? now.getFullYear() : now.getFullYear() - 1;
}

/**
 * Anno di corso dalla coorte, tenendo conto che l'anno accademico parte a settembre
 * (con il calcolo sull'anno solare, a gennaio un iscritto del 2025 risultava gia' al 2° anno)
 */
function calculateCourseYear(cohortYear, maxYears = MAX_COURSE_YEARS, now = new Date()) {
  if (!cohortYear) return 1;
  const year = currentAcademicStart(now) - Number(cohortYear) + 1;
  return Math.min(Math.max(year, 1), maxYears);
}

/**
 * Canale dall'iniziale del cognome; null se il cognome non c'e' (nessun valore di ripiego)
 */
function determinaCanale(cognome, regole = DEFAULT_CANALI) {
  const initial = String(cognome || '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .trim()
    .charAt(0)
    .toUpperCase();
  if (!initial) return null;
  const rule = regole.find(item => initial >= item.da && initial <= item.a);
  return rule ? rule.canale : null;
}

/**
 * Una coorte e' plausibile se non e' nel futuro e non e' di oltre 8 anni fa
 */
function sanitizeCohort(value, now = new Date()) {
  const year = Number(value);
  if (!Number.isInteger(year)) return null;
  const start = currentAcademicStart(now);
  return year >= start - 8 && year <= start ? year : null;
}

/**
 * Stima della coorte dal primo anno accademico che compare nei versamenti
 * (es. "2025/2026"): la prima tassa si paga all'immatricolazione.
 */
function inferCohortFromTaxes(taxes, now = new Date()) {
  const items = [...(taxes?.paid || []), ...(taxes?.pending || [])];
  const years = [];
  for (const item of items) {
    const text = typeof item === 'string' ? item : JSON.stringify(item);
    for (const match of text.matchAll(/(20\d{2})\s*\/\s*(?:20)?\d{2}/g)) {
      years.push(Number(match[1]));
    }
  }
  const valid = years.map(year => sanitizeCohort(year, now)).filter(Boolean);
  return valid.length ? Math.min(...valid) : null;
}

function cohortFromCourseYear(courseYear, now = new Date()) {
  const year = Number(courseYear);
  if (!Number.isInteger(year) || year < 1 || year > MAX_COURSE_YEARS) return null;
  return sanitizeCohort(currentAcademicStart(now) - year + 1, now);
}

/**
 * Maiuscolo, senza accenti, solo lettere e cifre separate da uno spazio
 */
function normalizeText(value) {
  return String(value ?? '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toUpperCase()
    .replace(/[^A-Z0-9]+/g, ' ')
    .trim();
}

/**
 * Tra i piani del database (listPlans(cohortYear) -> [{ course_id, course_code, course_name }]) cerca
 * quello del corso. Corrispondenza esatta di nome o codice prima; se resta un solo candidato lo usa,
 * se sono piu' d'uno non indovina (es. triennale e magistrale con nomi simili).
 */
function findPlanByName(courseName, codes, cohortYear, listPlans, findPlan) {
  const target = normalizeText(courseName);
  if (!target || typeof listPlans !== 'function') return null;
  const wantedCodes = new Set(codes.map(normalizeText).filter(Boolean));

  const plans = listPlans(cohortYear) || [];
  const exact = [];
  const partial = [];
  for (const plan of plans) {
    const name = normalizeText(plan.course_name);
    const planCodes = [plan.course_code, plan.course_id].map(normalizeText).filter(Boolean);
    if (name === target || planCodes.some(code => wantedCodes.has(code))) exact.push(plan);
    else if (name.length >= 8 && (name.includes(target) || target.includes(name))) partial.push(plan);
  }
  const pool = exact.length ? exact : partial;
  if (pool.length !== 1) return null;
  const chosen = String(pool[0].course_id);
  return findPlan(chosen, cohortYear) ? chosen : null;
}

/**
 * Costruisce il profilo dai dati letti su Segrepass.
 *  - nessun valore inventato: se un dato non si ricava resta null (e la UI lo chiede una volta sola);
 *  - coorte: dato esplicito di Segrepass > profilo gia' salvato > stima dai versamenti > anno di corso;
 *  - corso: si prova ogni candidato (codice, nome) contro i manifesti presenti nel database con
 *    findPlan(courseId, cohortYear); se nessuno coincide si cerca per nome tra i piani (listPlans),
 *    cosi' "scritto diversamente" non rompe piu' la ricerca e un id sbagliato salvato si corregge da solo;
 *  - canale: se il profilo ne ha gia' uno (anche impostato a mano) resta quello.
 *
 * existing: profilo gia' salvato (colonne snake_case); findPlan: (courseId, cohortYear) => piano | null
 */
function buildStudentProfile(anagrafica = {}, { taxes = null, existing = null, findPlan = null, listPlans = null, now = new Date() } = {}) {
  // Maiuscolo e spazi singoli: "Ingegneria  Informatica" e "INGEGNERIA INFORMATICA" sono lo stesso corso
  const courseName = (anagrafica.corsoDiLaurea ? String(anagrafica.corsoDiLaurea).replace(/\s+/g, ' ').trim().toUpperCase() : null)
    || existing?.course_name
    || null;

  const cohortYear = sanitizeCohort(anagrafica.annoImmatricolazione, now)
    ?? sanitizeCohort(existing?.cohort_year, now)
    ?? inferCohortFromTaxes(taxes, now)
    ?? cohortFromCourseYear(anagrafica.annoCorso, now);

  const normalized = courseName && typeof normalizeName === 'function' ? normalizeName(courseName) : courseName;
  const candidates = [...new Set(
    [
      anagrafica.codiceCorso,
      existing?.course_id,
      courseName && COURSE_CODE_MAP[courseName],
      normalized,
      courseName
    ]
      .filter(Boolean)
      .flatMap(value => [String(value).trim(), String(value).trim().toUpperCase()])
  )];

  let courseId = null;
  if (typeof findPlan === 'function' && cohortYear) {
    for (const candidate of candidates) {
      const plan = findPlan(candidate, cohortYear);
      if (plan) {
        courseId = String(plan.course_id ?? candidate);
        break;
      }
    }
  }
  // Nessun candidato coincide con un manifesto: si cerca per nome tra i piani presenti nel database
  if (!courseId && typeof findPlan === 'function' && cohortYear && courseName) {
    courseId = findPlanByName(courseName, [anagrafica.codiceCorso, existing?.course_id].filter(Boolean), cohortYear, listPlans, findPlan);
  }
  courseId = courseId ?? existing?.course_id ?? candidates[0] ?? null;

  const surname = anagrafica.cognome ? String(anagrafica.cognome).trim().toUpperCase() : '';
  const surnameInitial = surname.charAt(0) || existing?.surname_initial || null;
  const channel = existing?.channel || determinaCanale(surname || existing?.surname_initial);

  const currentCourseYear = cohortYear
    ? calculateCourseYear(cohortYear, MAX_COURSE_YEARS, now)
    : (existing?.current_course_year ?? null);

  return { courseId, courseName, cohortYear, currentCourseYear, channel, surnameInitial };
}

 * Compatibilita' con il vecchio nome: stesso risultato, ma senza valori inventati
 */
function extractStudentProfile(anagrafica = {}, options = {}) {
  return buildStudentProfile(anagrafica, options);
}

const { getDatiCarriera } = require('../scraper/segrepass');

const statuses = new Map();
const inFlight = new Set();

// Dopo un errore non si ritenta a ogni poll: si evita di martellare il portale
// (e di far risultare offline i nodi Segrepass) con login ripetuti.
const RETRY_COOLDOWN_MS = 2 * 60 * 1000;

function getSlowStatus(profileId) {
  return statuses.get(profileId) || { state: 'idle' };
}

function resetSlowSync(profileId) {
  statuses.delete(profileId);
  snapshots.setSlowSnapshot(profileId, null);
}

/**
 * Carica tasse e appelli in background. Non blocca il login: la UI mostra
 * subito esami/profilo e interroga /api/career/slow-data per il resto.
 */
function startSlowSync(profileId, credentials, dependencies = {}) {
  if (inFlight.has(profileId)) return getSlowStatus(profileId);
  if (snapshots.getSlowSnapshot(profileId)) return getSlowStatus(profileId);

  const scraper = dependencies.getDatiCarriera || getDatiCarriera;
  inFlight.add(profileId);
  statuses.set(profileId, { state: 'loading', startedAt: Date.now() });

  Promise.resolve()
    .then(() => scraper(credentials, { phase: 'slow' }))
    .then(data => {
      const taxes = data?.tasse || null;
      const prenotazioni = Array.isArray(data?.prenotazioni) ? data.prenotazioni : [];

      snapshots.setTaxesSnapshot(profileId, taxes);
      snapshots.setSlowSnapshot(profileId, {
        taxes,
        prenotazioni,
        updatedAt: Date.now()
      });
      statuses.set(profileId, { state: 'ready', finishedAt: Date.now() });
    })
    .catch(error => {
      console.warn(`[slow-sync] Dati lenti non disponibili: ${error?.message || error}`);
      statuses.set(profileId, {
        state: 'error',
        error: error?.code || 'SLOW_SYNC_FAILED',
        finishedAt: Date.now()
      });
    })
    .finally(() => inFlight.delete(profileId));

  return getSlowStatus(profileId);
}

/**
 * Avvia il caricamento se non c'è già un risultato o un job in corso.
 * Utile dopo un riavvio del server: le credenziali sono ancora in memoria.
 */
function ensureSlowSync(profileId, credentials, dependencies = {}) {
  const status = getSlowStatus(profileId);
  if (status.state === 'loading') return status;
  if (snapshots.getSlowSnapshot(profileId)) return status;
  if (status.state === 'error' && Date.now() - (status.finishedAt || 0) < RETRY_COOLDOWN_MS) return status;
  return startSlowSync(profileId, credentials, dependencies);
}

function clearSlowStatus(profileId) {
  statuses.delete(profileId);
}

  ensureSlowSync,
  getSlowStatus,
  resetSlowSync,
  clearSlowStatus

function execute(overrides = {}) {
  const req = { method: 'POST', headers: {}, protocol: 'http', ...overrides };

===== FILE: utils\sessionStore.js =====
// Credenziali Segrepass tenute solo in memoria, per la durata della sessione.
// Non vengono mai inserite nel token né restituite al client.
const DEFAULT_TTL_MS = 12 * 60 * 60 * 1000;
const credentials = new Map();

function setCredentials(profileId, { username, password } = {}, ttlMs = DEFAULT_TTL_MS) {
  if (!profileId || !username || !password) return false;
  credentials.set(profileId, { username, password, expiresAt: Date.now() + ttlMs });
  return true;
}

function getCredentials(profileId) {
  const entry = credentials.get(profileId);
  if (!entry) return null;
  if (entry.expiresAt <= Date.now()) {
    credentials.delete(profileId);
    return null;
  }
  return { username: entry.username, password: entry.password };
}

function deleteCredentials(profileId) {
  credentials.delete(profileId);
}

function clearCredentials() {
  credentials.clear();
}

module.exports = { setCredentials, getCredentials, deleteCredentials, clearCredentials };
