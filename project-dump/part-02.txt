const crypto = require('node:crypto');
const { JWT_SECRET } = require('../lib/config');

const DEFAULT_TTL_MS = 15 * 60 * 1000;
const cache = new Map();
const inFlight = new Map();

function ttlMs() {
  const configured = Number(process.env.CAREER_CACHE_TTL_MS);
  return Number.isFinite(configured) && configured >= 0 ? configured : DEFAULT_TTL_MS;
}

/**
 * Chiave per studente + credenziale. Della password resta solo un HMAC:
 * una password errata non può quindi riutilizzare i dati in cache.
 */
function cacheKey(profileId, password) {
  const fingerprint = crypto
    .createHmac('sha256', JWT_SECRET)
    .update(String(password || ''))
    .digest('hex');
  return `${profileId}:${fingerprint}`;
}

function prune(now = Date.now()) {
  const ttl = ttlMs();
  for (const [key, entry] of cache) {
    if (now - entry.storedAt > ttl) cache.delete(key);
  }
}

function getCachedCareer(profileId, password, now = Date.now()) {
  const ttl = ttlMs();
  if (ttl <= 0) return null;

  const key = cacheKey(profileId, password);
  const entry = cache.get(key);
  if (!entry) return null;
  if (now - entry.storedAt > ttl) {
    cache.delete(key);
    return null;
  }
  return { value: entry.value, storedAt: entry.storedAt };
}

function setCachedCareer(profileId, password, value, now = Date.now()) {
  if (ttlMs() <= 0) return;
  const key = cacheKey(profileId, password);
  cache.set(key, { value, storedAt: now, profileId });
  prune(now);
}

/**
 * Restituisce i dati di carriera dalla cache oppure esegue il loader una sola
 * volta, anche se arrivano più login contemporanei per lo stesso studente.
 */
async function getOrLoadCareer(profileId, password, loader) {
  const cached = getCachedCareer(profileId, password);
  if (cached) return { value: cached.value, fromCache: true };

  const key = cacheKey(profileId, password);
  if (inFlight.has(key)) {
    return { value: await inFlight.get(key), fromCache: false, shared: true };
  }

  const task = Promise.resolve()
    .then(loader)
    .then(value => {
      setCachedCareer(profileId, password, value);
      return value;
    })
    .finally(() => inFlight.delete(key));

  inFlight.set(key, task);
  return { value: await task, fromCache: false };
}

function clearCareerCache(profileId) {
  if (!profileId) {
    cache.clear();
    inFlight.clear();
    return;
  }
  for (const key of cache.keys()) {
    if (key.startsWith(`${profileId}:`)) cache.delete(key);
  }
}

module.exports = { getCachedCareer, setCachedCareer, getOrLoadCareer, clearCareerCache };

// Trasforma le righe della tabella "Esami Sostenuti" di Segrepass in oggetti Esame
// (stesso formato di api.ts) e calcola CFU, medie e base di laurea.

const normalizeKey = value => String(value ?? '')
  .normalize('NFD')
  .replace(/[\u0300-\u036f]/g, '')
  .toLowerCase()
  .trim();

const cleanText = value => String(value ?? '').replace(/\s+/g, ' ').trim();

/**
 * Primo valore non vuoto la cui intestazione corrisponde a uno dei pattern (in ordine di priorita')
 */
function pickField(record, patterns, exclude = null) {
  const entries = Object.entries(record);
  for (const pattern of patterns) {
    for (const [key, value] of entries) {
      const normalized = normalizeKey(key);
      if (exclude && exclude.test(normalized)) continue;
      if (pattern.test(normalized) && cleanText(value)) return cleanText(value);
    }
  }
  return null;
}

/**
 * "28", "30 e lode", "30L", "28/30" -> voto; "IDONEO" -> giudizio
 */
function parseGrade(raw) {
  const text = cleanText(raw).toUpperCase();
  if (!text) return { voto: null, lode: false, giudizio: null };
  const digits = text.match(/(\d{2})/);
  const voto = digits ? Number(digits[1]) : null;
  if (voto !== null && voto >= 18 && voto <= 30) {
    return { voto, lode: voto === 30 && /LODE|30\s*L\b/.test(text), giudizio: null };
  }
  return { voto: null, lode: false, giudizio: cleanText(raw) };
}

const PASSED_JUDGMENT = /idone|approvat|superat|positiv|conseguit|riconosciut|convalid/i;

/**
 * Righe del libretto -> Esame[] (solo esami superati)
 */
function normalizeEsami(records = []) {
  const hasGradeColumn = records.some(record =>
    Object.keys(record).some(key => /voto|valutazione|esito|giudizio/.test(normalizeKey(key)))
  );

  const exams = [];
  for (const record of records) {
    const name = pickField(
      record,
      [/insegnamento/, /denominazione/, /attivita/, /descrizione/, /materia/, /^esame/],
      /codice|^cod$/
    );
    if (!name) continue;

    const gradeRaw = pickField(record, [/^voto/, /valutazione/, /esito/, /giudizio/]);
    const { voto, lode, giudizio } = parseGrade(gradeRaw);
    // Senza colonna dei voti nella tabella, tutte le righe sono considerate superate
    const passed = !hasGradeColumn || voto !== null || (giudizio && PASSED_JUDGMENT.test(giudizio));
    if (!passed) continue;

    const cfuRaw = pickField(record, [/cfu/, /crediti/, /peso/]);
    const cfu = cfuRaw ? Number((cfuRaw.replace(',', '.').match(/\d+(?:\.\d+)?/) || ['0'])[0]) : 0;

    exams.push({
      materia: name,
      insegnamento: name,
      cfu: Number.isFinite(cfu) ? cfu : 0,
      voto,
      giudizio: voto === null ? giudizio : null,
      lode,
      data: pickField(record, [/data/, /verbalizz/, /superamento/]) || undefined
    });
  }
  return exams;
}

/**
 * Metriche nel formato atteso dal frontend (tipo Metriche di api.ts)
 */
function buildMetriche(records = [], { cfuTotali = 180 } = {}) {
  const esami = normalizeEsami(records);
  const graded = esami.filter(exam => exam.voto !== null);

  const cfuAcquisiti = esami.reduce((sum, exam) => sum + exam.cfu, 0);
  const gradedCfu = graded.reduce((sum, exam) => sum + exam.cfu, 0);

  const mediaAritmetica = graded.length
    ? graded.reduce((sum, exam) => sum + exam.voto, 0) / graded.length
    : null;
  const mediaPonderata = gradedCfu > 0
    ? graded.reduce((sum, exam) => sum + exam.voto * exam.cfu, 0) / gradedCfu
    : null;

  const format = value => (value === null ? null : value.toFixed(2));
  return {
    esami,
    cfuAcquisiti,
    cfuTotali,
    avanzamentoCFU: Math.min(100, Math.round((cfuAcquisiti / cfuTotali) * 100)),
    mediaAritmetica: format(mediaAritmetica),
    mediaPonderata: format(mediaPonderata),
    baseLaurea: mediaPonderata === null ? null : format((mediaPonderata * 110) / 30)
  };
}

module.exports = { buildMetriche, normalizeEsami, parseGrade };

const TTL_MS = 12 * 60 * 60 * 1000; // 12 ore di validità per la sessione in memoria

const careerSnapshots = new Map();
const docentiSnapshots = new Map();
const taxesSnapshots = new Map(); // Aggiunto store per le tasse
const slowSnapshots = new Map(); // Tasse + appelli caricati in background
const docentiLoading = new Set();

function prune(store) {
  const now = Date.now();
  for (const [key, entry] of store) {
    if (now - entry.storedAt > TTL_MS) store.delete(key);
  }
}

function readFresh(store, profileId) {
  const entry = store.get(profileId);
  if (!entry) return null;
  if (Date.now() - entry.storedAt > TTL_MS) {
    store.delete(profileId);
    return null;
  }
  return entry.value;
}

// -- Esami --
function setCareerSnapshot(profileId, esami) {
  prune(careerSnapshots);
  careerSnapshots.set(profileId, { value: Array.isArray(esami) ? esami : [], storedAt: Date.now() });
}

function getCareerSnapshot(profileId) {
  return readFresh(careerSnapshots, profileId);
}

// -- Tasse (Nuovo) --
function setTaxesSnapshot(profileId, data) {
  prune(taxesSnapshots);
  taxesSnapshots.set(profileId, { value: data || null, storedAt: Date.now() });
}

function getTaxesSnapshot(profileId) {
  return readFresh(taxesSnapshots, profileId);
}

// -- Dati lenti (tasse + appelli) --
function setSlowSnapshot(profileId, data) {
  prune(slowSnapshots);

// Cache breve per i dati del portale Docenti: evita di riaprire un browser
// a ogni chiamata del frontend (dashboard + annunci + iscrizioni + appelli).
const crypto = require('node:crypto');
const { getDocentiSnapshot, isDocentiLoading } = require('./career-snapshots');
const { JWT_SECRET } = require('../lib/config');

const TTL_MS = 5 * 60 * 1000;
// Il login avvia l'acquisizione in background: le richieste che arrivano nel
// frattempo attendono lo stesso job invece di avviare altri browser.
const LOADING_WAIT_MS = 60000;
const LOADING_POLL_MS = 500;
const cache = new Map();
const inFlight = new Map();

function delay(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

function credentialFingerprint(credentials) {
  if (!credentials) return '';
  const username = credentials.username || credentials.codiceFiscale || '';
  const password = credentials.password || '';
  if (!username && !password) return '';
  return crypto
    .createHmac('sha256', JWT_SECRET)
    .update(`${username}\u0000${password}`)
    .digest('hex')
    .slice(0, 16);
}

function cacheKey(profileId, credentials) {
  const fingerprint = credentialFingerprint(credentials);
  return fingerprint ? `${profileId}:${fingerprint}` : String(profileId);
}

async function waitForBackground(profileId) {
  const deadline = Date.now() + LOADING_WAIT_MS;
  while (Date.now() < deadline) {
    const snapshot = getDocentiSnapshot(profileId);
    if (snapshot) return snapshot;
    if (!isDocentiLoading(profileId)) return null;
    await delay(LOADING_POLL_MS);
  }
  return getDocentiSnapshot(profileId);
}

function normalizeArgs(profileId, second, third, fourth) {
  if (typeof second === 'function') {
    return { credentials: fourth || third || null, loader: second };
  }
  return { credentials: second || null, loader: third };
}

async function getDocentiCached(profileId, second, third, fourth) {
  const { credentials, loader } = normalizeArgs(profileId, second, third, fourth);
  const key = cacheKey(profileId, credentials);

  const entry = cache.get(key);
  if (entry && Date.now() - entry.storedAt < TTL_MS) return entry.value;

  // Il login carica già i dati in background: se ci sono, usali subito.
  const background = getDocentiSnapshot(profileId);
  if (background) {
    cache.set(key, { value: background, storedAt: Date.now() });
    return background;
  }

  if (isDocentiLoading(profileId)) {
    const loaded = await waitForBackground(profileId);
    if (loaded) {
      cache.set(key, { value: loaded, storedAt: Date.now() });
      return loaded;
    }
  }

  if (inFlight.has(key)) return inFlight.get(key);

  const task = Promise.resolve()
    .then(loader)
    .then(value => {
      cache.set(key, { value, storedAt: Date.now() });
      return value;
    })
    .finally(() => inFlight.delete(key));

  inFlight.set(key, task);
  return task;
}

function clearDocentiCache(profileId) {
  if (profileId) {
    const prefix = String(profileId);
    for (const key of [...cache.keys()]) {
      if (key === prefix || key.startsWith(`${prefix}:`)) cache.delete(key);
    }
    for (const key of [...inFlight.keys()]) {
      if (key === prefix || key.startsWith(`${prefix}:`)) inFlight.delete(key);
    }
  } else {
    cache.clear();
    inFlight.clear();
  }
}

module.exports = { getDocentiCached, clearDocentiCache, TTL_MS, cacheKey };

function parsePopupText(rawText) {
  const text = String(rawText || '').replace(/\u00a0/g, ' ').replace(/\r/g, '');
  const between = (start, end) => {
    const startAt = text.indexOf(start);
    if (startAt < 0) return '';
    const valueStart = startAt + start.length;
    const endAt = text.indexOf(end, valueStart);
    return text.slice(valueStart, endAt < 0 ? text.length : endAt).trim();
  };

  const rawName = between('Nome insegnamento', 'Argomento lezione');
  const teacherText = between('Docenti', 'Data e Ora');
  const dateTime = between('Data e Ora', 'Aula').replace(/\s+/g, ' ').trim();
  const room = (text.match(/Nome aula:\s*([\s\S]*?)(?=Capacità:|Sede:|Google Maps|$)/i)?.[1] || '').trim();
  const campus = (text.match(/Sede:\s*([^\[]+)/i)?.[1] || '').trim();
  const coursePath = between('Percorso didattico', 'Numero eventi collegati').replace(/\s+/g, ' ').trim();
  const dateMatch = dateTime.match(/(?:\w+\s+)?(\d{1,2})\s+([a-zà-ù]+)\s+(\d{4})\s*(\d{2}:\d{2})\s*-\s*(\d{2}:\d{2})/i);
  const month = dateMatch ? MONTHS[normalizeLabel(dateMatch[2])] : null;

  return {
    courseName: rawName.replace(/\s*\([^)]*\)\s*$/, '').trim(),
    channel: rawName.match(/\(([^()]*)\)\s*$/)?.[1] || null,
    teachers: teacherText.split('\n').map(name => name.trim()).filter(Boolean),
    dateTime,
    date: dateMatch && month ? `${dateMatch[3]}-${month}-${dateMatch[1].padStart(2, '0')}` : null,
    startTime: dateMatch?.[4] || null,
    endTime: dateMatch?.[5] || null,
    room: room.replace(/\s+/g, ' ').trim() || null,
    campus: campus || null,
    coursePath: coursePath || null
  };
}

function getCourseOptions(options, { courseId, courseName, studyYear }) {
  const wantedName = normalizeCourseName(String(courseName || '')
    .replace(/\[[^\]]+\]/g, '')
    .replace(/\bL-?\s*8\b/i, ''));
  const candidates = options.map(option => {
    const text = String(option.text || '').trim();
    const isThirdYear = /terzo anno/i.test(text);
    const displayName = normalizeCourseName(text.replace(/\s*\([^)]*\)/g, '').replace(/\s*[-–]\s*terzo anno/i, ''));
    const codeMatch = Boolean(courseId && String(option.value).toLowerCase() === String(courseId).toLowerCase());
    const similarity = wantedName ? nameSimilarity(displayName, wantedName) : 0;
    const yearMatch = (studyYear >= 3) === isThirdYear;
    return { option, text, codeMatch, similarity, score: similarity + (yearMatch ? 0.25 : 0) + (codeMatch ? 0.1 : 0) };
  }).filter(candidate => candidate.option.value && (candidate.codeMatch || candidate.similarity >= 0.72));

  candidates.sort((left, right) => right.score - left.score);
  return candidates[0] || null;
}

function extractGroupRange(text) {
  const token = String(text || '').toUpperCase().match(/\b([A-Z])\s*[-–]\s*([A-Z]+)/);
  if (!token) return null;
  return { first: token[1], last: token[2][0] };
}

function matchesStudyGroup(optionText, channel, surnameInitial) {
  const normalizedOption = normalizeLabel(optionText);
  const channelToken = normalizeLabel(String(channel || '').split(':').pop().replace(/corso generico/i, ''));
  if (channelToken && normalizedOption.includes(channelToken)) return true;

  const channelPart = String(surnameInitial || channel || '').toUpperCase().split(':').pop();
  const initial = channelPart.match(/[A-Z]/)?.[0];
  const range = extractGroupRange(optionText);
  return Boolean(initial && range && initial >= range.first && initial <= range.last);
}

function selectStudyGroup(options, studyYear, channel, surnameInitial) {
  const relevant = options.filter(option => new RegExp(`^\\s*${studyYear}\\s*[-–]`).test(option.text));
  const matches = relevant.filter(option => matchesStudyGroup(option.text, channel, surnameInitial));
  if (matches.length === 1) return matches[0];
  if (matches.length > 1) {
    const exact = matches.find(option => normalizeLabel(option.text).includes(normalizeLabel(channel || '')));
    if (exact) return exact;
  }
  return null;
}

function cacheKey(filters) {
  return JSON.stringify({
    courseId: filters.courseId,
    courseName: normalizeLabel(filters.courseName),
    academicYear: filters.academicYear,
    studyYear: filters.studyYear,
    semester: filters.semester,
    channel: normalizeLabel(filters.channel),
    surnameInitial: normalizeLabel(filters.surnameInitial)
  });
}

function readCache(key, now = Date.now()) {
  const cached = timetableCache.get(key);
  if (!cached || cached.expiresAt <= now) {
    timetableCache.delete(key);
    return null;
  }
  return cached.value;
}

function writeCache(key, value, now = Date.now()) {
  for (const [entryKey, entry] of timetableCache) {
    if (entry.expiresAt <= now) timetableCache.delete(entryKey);
  }
  timetableCache.set(key, { value, expiresAt: now + CACHE_TTL_MS });
}

// Solo DE3 e' verificato (compare cosi' sul portale Docenti: "DE3 INGEGNERIA INFORMATICA").
// Per gli altri corsi si usa il codice letto da Segrepass oppure il nome normalizzato:
// i codici DE1, DE2, DE4 e DE5 della versione precedente erano inventati e sono stati tolti.
const COURSE_CODE_MAP = {
  'INGEGNERIA INFORMATICA': 'DE3'
};

// ATTENZIONE: solo "A-D" e' confermato dal tuo profilo. Gli altri intervalli sono un'ipotesi:
// confrontali con i canali che mostra EasyAcademy per il tuo corso.
const DEFAULT_CANALI = [
  { da: 'A', a: 'D', canale: 'A-D' },
  { da: 'E', a: 'O', canale: 'E-O' },
  { da: 'P', a: 'Z', canale: 'P-Z' }
];

const MAX_COURSE_YEARS = 3;

/**
 * Anno di inizio dell'anno accademico in corso (l'anno accademico parte a settembre)
 */
function currentAcademicStart(now = new Date()) {
  return now.getMonth() >= 8 ? now.getFullYear() : now.getFullYear() - 1;
}

/**
 * Anno di corso dalla coorte, tenendo conto che l'anno accademico parte a settembre
 * (con il calcolo sull'anno solare, a gennaio un iscritto del 2025 risultava gia' al 2° anno)
 */
function calculateCourseYear(cohortYear, maxYears = MAX_COURSE_YEARS, now = new Date()) {
  if (!cohortYear) return 1;
  const year = currentAcademicStart(now) - Number(cohortYear) + 1;
  return Math.min(Math.max(year, 1), maxYears);
}

/**
 * Canale dall'iniziale del cognome; null se il cognome non c'e' (nessun valore di ripiego)
 */
function determinaCanale(cognome, regole = DEFAULT_CANALI) {
  const initial = String(cognome || '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .trim()
    .charAt(0)
    .toUpperCase();
  if (!initial) return null;
  const rule = regole.find(item => initial >= item.da && initial <= item.a);
  return rule ? rule.canale : null;
}

/**
 * Una coorte e' plausibile se non e' nel futuro e non e' di oltre 8 anni fa
 */
function sanitizeCohort(value, now = new Date()) {
  const year = Number(value);
  if (!Number.isInteger(year)) return null;
  const start = currentAcademicStart(now);
  return year >= start - 8 && year <= start ? year : null;
}

/**
 * Stima della coorte dal primo anno accademico che compare nei versamenti
 * (es. "2025/2026"): la prima tassa si paga all'immatricolazione.
 */
function inferCohortFromTaxes(taxes, now = new Date()) {
  const items = [...(taxes?.paid || []), ...(taxes?.pending || [])];
  const years = [];
  for (const item of items) {
    const text = typeof item === 'string' ? item : JSON.stringify(item);
    for (const match of text.matchAll(/(20\d{2})\s*\/\s*(?:20)?\d{2}/g)) {
      years.push(Number(match[1]));
    }
  }
  const valid = years.map(year => sanitizeCohort(year, now)).filter(Boolean);
  return valid.length ? Math.min(...valid) : null;
}

function cohortFromCourseYear(courseYear, now = new Date()) {
  const year = Number(courseYear);
  if (!Number.isInteger(year) || year < 1 || year > MAX_COURSE_YEARS) return null;
  return sanitizeCohort(currentAcademicStart(now) - year + 1, now);
}

/**
 * Maiuscolo, senza accenti, solo lettere e cifre separate da uno spazio
 */
function normalizeText(value) {
  return String(value ?? '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toUpperCase()
    .replace(/[^A-Z0-9]+/g, ' ')
    .trim();
}

/**
 * Tra i piani del database (listPlans(cohortYear) -> [{ course_id, course_code, course_name }]) cerca
 * quello del corso. Corrispondenza esatta di nome o codice prima; se resta un solo candidato lo usa,
 * se sono piu' d'uno non indovina (es. triennale e magistrale con nomi simili).
 */
function findPlanByName(courseName, codes, cohortYear, listPlans, findPlan) {
  const target = normalizeText(courseName);
  if (!target || typeof listPlans !== 'function') return null;
  const wantedCodes = new Set(codes.map(normalizeText).filter(Boolean));

  const plans = listPlans(cohortYear) || [];
  const exact = [];
  const partial = [];
  for (const plan of plans) {
    const name = normalizeText(plan.course_name);
    const planCodes = [plan.course_code, plan.course_id].map(normalizeText).filter(Boolean);
    if (name === target || planCodes.some(code => wantedCodes.has(code))) exact.push(plan);
    else if (name.length >= 8 && (name.includes(target) || target.includes(name))) partial.push(plan);
  }
  const pool = exact.length ? exact : partial;
  if (pool.length !== 1) return null;
  const chosen = String(pool[0].course_id);
  return findPlan(chosen, cohortYear) ? chosen : null;
}

/**
 * Costruisce il profilo dai dati letti su Segrepass.
 *  - nessun valore inventato: se un dato non si ricava resta null (e la UI lo chiede una volta sola);
 *  - coorte: dato esplicito di Segrepass > profilo gia' salvato > stima dai versamenti > anno di corso;
 *  - corso: si prova ogni candidato (codice, nome) contro i manifesti presenti nel database con
 *    findPlan(courseId, cohortYear); se nessuno coincide si cerca per nome tra i piani (listPlans),
 *    cosi' "scritto diversamente" non rompe piu' la ricerca e un id sbagliato salvato si corregge da solo;
 *  - canale: se il profilo ne ha gia' uno (anche impostato a mano) resta quello.
 *
 * existing: profilo gia' salvato (colonne snake_case); findPlan: (courseId, cohortYear) => piano | null
 */
function buildStudentProfile(anagrafica = {}, { taxes = null, existing = null, findPlan = null, listPlans = null, now = new Date() } = {}) {
  // Maiuscolo e spazi singoli: "Ingegneria  Informatica" e "INGEGNERIA INFORMATICA" sono lo stesso corso
  const courseName = (anagrafica.corsoDiLaurea ? String(anagrafica.corsoDiLaurea).replace(/\s+/g, ' ').trim().toUpperCase() : null)
    || existing?.course_name
    || null;

  const cohortYear = sanitizeCohort(anagrafica.annoImmatricolazione, now)
    ?? sanitizeCohort(existing?.cohort_year, now)
    ?? inferCohortFromTaxes(taxes, now)
    ?? cohortFromCourseYear(anagrafica.annoCorso, now);

  const normalized = courseName && typeof normalizeName === 'function' ? normalizeName(courseName) : courseName;
  const candidates = [...new Set(
    [
      anagrafica.codiceCorso,
      existing?.course_id,
      courseName && COURSE_CODE_MAP[courseName],
      normalized,
      courseName
    ]
      .filter(Boolean)
      .flatMap(value => [String(value).trim(), String(value).trim().toUpperCase()])
  )];

  let courseId = null;
  if (typeof findPlan === 'function' && cohortYear) {
    for (const candidate of candidates) {
      const plan = findPlan(candidate, cohortYear);
      if (plan) {
        courseId = String(plan.course_id ?? candidate);
        break;
      }
    }
  }
  // Nessun candidato coincide con un manifesto: si cerca per nome tra i piani presenti nel database
  if (!courseId && typeof findPlan === 'function' && cohortYear && courseName) {
    courseId = findPlanByName(courseName, [anagrafica.codiceCorso, existing?.course_id].filter(Boolean), cohortYear, listPlans, findPlan);
  }
  courseId = courseId ?? existing?.course_id ?? candidates[0] ?? null;

  const surname = anagrafica.cognome ? String(anagrafica.cognome).trim().toUpperCase() : '';
  const surnameInitial = surname.charAt(0) || existing?.surname_initial || null;
  const channel = existing?.channel || determinaCanale(surname || existing?.surname_initial);

  const currentCourseYear = cohortYear
    ? calculateCourseYear(cohortYear, MAX_COURSE_YEARS, now)
    : (existing?.current_course_year ?? null);

  return { courseId, courseName, cohortYear, currentCourseYear, channel, surnameInitial };
}

 */

  return buildStudentProfile(anagrafica, options);
}

const RETRY_COOLDOWN_MS = 2 * 60 * 1000;

function getSlowStatus(profileId) {
  return statuses.get(profileId) || { state: 'idle' };
}

function resetSlowSync(profileId) {
  statuses.delete(profileId);
  snapshots.setSlowSnapshot(profileId, null);
}

/**
 * Carica tasse e appelli in background. Non blocca il login: la UI mostra
 * subito esami/profilo e interroga /api/career/slow-data per il resto.
 */
function startSlowSync(profileId, credentials, dependencies = {}) {
  if (inFlight.has(profileId)) return getSlowStatus(profileId);
  if (snapshots.getSlowSnapshot(profileId)) return getSlowStatus(profileId);

  const scraper = dependencies.getDatiCarriera || getDatiCarriera;
  inFlight.add(profileId);
  statuses.set(profileId, { state: 'loading', startedAt: Date.now() });

  Promise.resolve()
    .then(() => scraper(credentials, { phase: 'slow' }))
    .then(data => {
      const taxes = data?.tasse || null;
      const prenotazioni = Array.isArray(data?.prenotazioni) ? data.prenotazioni : [];

      snapshots.setTaxesSnapshot(profileId, taxes);
      snapshots.setSlowSnapshot(profileId, {
        taxes,
        prenotazioni,
        updatedAt: Date.now()
      });
      statuses.set(profileId, { state: 'ready', finishedAt: Date.now() });
    })
    .catch(error => {
      console.warn(`[slow-sync] Dati lenti non disponibili: ${error?.message || error}`);
      statuses.set(profileId, {
        state: 'error',
        error: error?.code || 'SLOW_SYNC_FAILED',
        finishedAt: Date.now()
      });
    })
    .finally(() => inFlight.delete(profileId));

  return getSlowStatus(profileId);
}

/**
 * Avvia il caricamento se non c'è già un risultato o un job in corso.
 * Utile dopo un riavvio del server: le credenziali sono ancora in memoria.
 */
function ensureSlowSync(profileId, credentials, dependencies = {}) {
  const status = getSlowStatus(profileId);
  if (status.state === 'loading') return status;
  if (snapshots.getSlowSnapshot(profileId)) return status;
  if (status.state === 'error' && Date.now() - (status.finishedAt || 0) < RETRY_COOLDOWN_MS) return status;
  return startSlowSync(profileId, credentials, dependencies);
}

function clearSlowStatus(profileId) {
  statuses.delete(profileId);
}

    if (plan?.course_name) profile.courseName = plan.course_name;
  }

  const storedProfile = upsertStudentProfile(database, {
    id: cfClean,
    ...profile,
    esamiSostenuti: esami,
    tasse
  });

  snapshots.setCareerSnapshot(cfClean, esami);
  snapshots.setTaxesSnapshot(cfClean, tasse);
  // Ogni login invalida i dati Docenti precedenti e forza un refresh completo.
  snapshots.setDocentiSnapshot(cfClean, null);
  clearDocentiCache(cfClean);
  slowSync.resetSlowSync(cfClean);

  // Tasse e appelli partono in background: il login non li attende.
  const ensureSlow = dependencies.ensureSlowSync || slowSync.ensureSlowSync;
  const slowStatus = ensureSlow(cfClean, { codiceFiscale: cfClean, password }, dependencies);

  // Il portale Docenti richiede l'email istituzionale (@studenti.unina.it),
  // non il codice fiscale. Quella inserita nel form ha la precedenza.
  const docentiEmail = normalizeEmail(email) || resolveDocentiEmail(storedProfile || profile, cfClean);

  // La password viaggia nel JWT in forma cifrata (campo "pw").
  const token = jwt.sign(
    { sub: cfClean, codiceFiscale: cfClean, email: docentiEmail || undefined, pw: seal(password) },
    jwtSecret,
    { expiresIn: JWT_TTL }
  );

  const configuredDocenti = docentiCredentials(dependencies);
  const credentials = dependencies.docentiCredentials
    || (configuredDocenti.username && configuredDocenti.password
      ? configuredDocenti
      : (docentiEmail ? { username: docentiEmail, password } : {}));

  if (credentials.username && credentials.password) {
    setCredentials(cfClean, credentials);
  }

  const docentiSnapshot = snapshots.getDocentiSnapshot(cfClean);
  const alreadyLoading = snapshots.isDocentiLoading(cfClean);
  if (!docentiSnapshot && !alreadyLoading) {
    loadDocentiInBackground(cfClean, { ...dependencies, docentiCredentials: credentials });
  }

  const docentiLoading = !docentiSnapshot && (alreadyLoading || Boolean(credentials.username && credentials.password));
  const docentiReason = docentiSnapshot
    ? (docentiSnapshot.available ? null : docentiSnapshot.reason || 'DOCENTI_SCRAPE_FAILED')
    : (docentiLoading ? null : 'DOCENTI_CREDENTIALS_REQUIRED');

  return {
    token,
    expiresIn: JWT_TTL,
    profileId: cfClean,
    profile: storedProfile || profile,
    onboardingRequired: !hasProfileSetup(storedProfile || profile),
    fromCache,
    slowSync: slowStatus.state,
    docentiLoading,
    docentiReason,
    metriche: datiCarriera
  };
}

function execute(overrides = {}) {
  const req = { method: 'POST', headers: {}, protocol: 'http', ...overrides };
  const result = { statusCode: 200, body: null, nextCalled: false };

  const res = {
    status(code) { result.statusCode = code; return this; },
    json(body) { result.body = body; return this; }
  };

  csrfProtection(req, res, () => { result.nextCalled = true; });

}

const test = require('node:test');
const assert = require('node:assert/strict');
const express = require('express');
const { createDocentiRouter } = require('../routes/docenti');

async function withDocentiApi(t, loadDocenti) {
  const app = express();
  app.use('/api/docenti', (req, _res, next) => {
    req.user = { sub: 'profile-test' };
    next();
  }, createDocentiRouter({ loadDocenti }));

  const server = app.listen(0, '127.0.0.1');
  await new Promise((resolve, reject) => {
    server.once('listening', resolve);
    server.once('error', reject);
  });
  t.after(() => new Promise(resolve => server.close(resolve)));

  const { port } = server.address();
  return path => fetch(`http://127.0.0.1:${port}/api/docenti${path}`);
}

// Credenziali Segrepass tenute solo in memoria, per la durata della sessione.
// Non vengono mai inserite nel token né restituite al client.
const DEFAULT_TTL_MS = 12 * 60 * 60 * 1000;
const credentials = new Map();

function setCredentials(profileId, { username, password } = {}, ttlMs = DEFAULT_TTL_MS) {
  if (!profileId || !username || !password) return false;
  credentials.set(profileId, { username, password, expiresAt: Date.now() + ttlMs });
  return true;
}

function getCredentials(profileId) {
  const entry = credentials.get(profileId);
  if (!entry) return null;
  if (entry.expiresAt <= Date.now()) {
    credentials.delete(profileId);
    return null;
  }
  return { username: entry.username, password: entry.password };
}

function deleteCredentials(profileId) {
  credentials.delete(profileId);
}

function clearCredentials() {
  credentials.clear();
}

module.exports = { setCredentials, getCredentials, deleteCredentials, clearCredentials };

let outputContent = '';

function walkDir(dir) {
  const files = fs.readdirSync(dir);
  for (const file of files) {
    const fullPath = path.join(dir, file);
    const stat = fs.statSync(fullPath);

    if (stat.isDirectory()) {
      if (!excludeDirs.includes(file)) {
        walkDir(fullPath);
      }
    } else {
      const ext = path.extname(file);
      if (includeExtensions.includes(ext) || file === '.env') {
        const relativePath = path.relative(process.cwd(), fullPath);
        try {
          const content = fs.readFileSync(fullPath, 'utf8');
          outputContent += `\n\n==================================================\n`;
          outputContent += `FILE: ${relativePath}\n`;
          outputContent += `==================================================\n\n`;
          outputContent += content;
          console.log(`[OK] Aggiunto: ${relativePath}`);
        } catch (err) {
          console.log(`[SKIP] Impossibile leggere: ${relativePath}`);
        }
      }
    }
  }
}

function normalizeName(value) {
  return String(value || '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, ' ')
    .trim();
}

function parsePlanCourse(row) {
  if (!row) return null;
  let prerequisites = [];
  try {
    prerequisites = JSON.parse(row.prerequisites_json || '[]');
  } catch {}
  return {
    ...row,
    courseCode: row.course_code,
    courseName: row.course_name,
    studyYear: row.study_year,
    courseType: row.course_type,
    electiveGroup: row.elective_group,
    electiveRequiredCfu: row.elective_required_cfu,
    prerequisites
  };
}

function getStudentProfile(db, profileId) {

function upsertStudentProfile(db, profile) {
  db.prepare(`
    INSERT INTO StudentProfile (id, course_id, course_name, cohort_year, current_course_year, channel, surname_initial)
    VALUES (@id, @courseId, @courseName, @cohortYear, @currentCourseYear, @channel, @surnameInitial)
    ON CONFLICT(id) DO UPDATE SET
      course_id = COALESCE(excluded.course_id, StudentProfile.course_id),
      course_name = COALESCE(excluded.course_name, StudentProfile.course_name),
      cohort_year = COALESCE(excluded.cohort_year, StudentProfile.cohort_year),
      current_course_year = COALESCE(excluded.current_course_year, StudentProfile.current_course_year),
      channel = COALESCE(excluded.channel, StudentProfile.channel),
      surname_initial = COALESCE(excluded.surname_initial, StudentProfile.surname_initial),
      updated_at = CURRENT_TIMESTAMP
  `).run({
    id: profile.id,
    courseId: profile.courseId || null,
    courseName: profile.courseName || null,
    cohortYear: profile.cohortYear || null,
    currentCourseYear: profile.currentCourseYear || null,
    channel: profile.channel || null,
    surnameInitial: profile.surnameInitial || null
  });
  return getStudentProfile(db, profile.id);
}

function updateStudentProfile(db, profileId, updates) {
  const columns = {
    courseId: 'course_id',
    courseName: 'course_name',
    cohortYear: 'cohort_year',
    currentCourseYear: 'current_course_year',
    channel: 'channel',
    surnameInitial: 'surname_initial'
  };
  const entries = Object.entries(columns).filter(([key]) => Object.hasOwn(updates, key));
  if (entries.length === 0) return getStudentProfile(db, profileId);

  const assignments = entries.map(([, column]) => `${column} = @${column}`).join(', ');
  const values = Object.fromEntries(entries.map(([key, column]) => [column, updates[key] ?? null]));
  db.prepare(`UPDATE StudentProfile SET ${assignments}, updated_at = CURRENT_TIMESTAMP WHERE id = @id`)
    .run({ ...values, id: profileId });
  return getStudentProfile(db, profileId);
}

function saveStudyPlan(db, plan, courses) {
  const save = db.transaction(() => {
    db.prepare(`
      INSERT INTO StudyPlans (course_id, cohort_year, course_name, version, source_name, source_url)
      VALUES (@courseId, @cohortYear, @courseName, @version, @sourceName, @sourceUrl)
      ON CONFLICT(course_id, cohort_year, version) DO UPDATE SET
        course_name = excluded.course_name,
        source_name = excluded.source_name,
        source_url = excluded.source_url,
        imported_at = CURRENT_TIMESTAMP
    `).run({
      courseId: plan.courseId,
      cohortYear: plan.cohortYear,
      courseName: plan.courseName,
      version: String(plan.version || '1'),
      sourceName: plan.sourceName || null,
      sourceUrl: plan.sourceUrl || null
    });

    const storedPlan = db.prepare(`
      SELECT * FROM StudyPlans
      WHERE course_id = ? AND cohort_year = ? AND version = ?
    `).get(plan.courseId, plan.cohortYear, String(plan.version || '1'));

    db.prepare('DELETE FROM PlanCourses WHERE plan_id = ?').run(storedPlan.id);
    const insertCourse = db.prepare(`
      INSERT INTO PlanCourses (
        plan_id, course_code, course_name, normalized_name, cfu, study_year, semester,
        course_type, elective_group, elective_required_cfu, prerequisites_json
      ) VALUES (
        @planId, @courseCode, @courseName, @normalizedName, @cfu, @studyYear, @semester,
        @courseType, @electiveGroup, @electiveRequiredCfu, @prerequisitesJson
      )
    `);

    for (const course of courses) {
      insertCourse.run({
        planId: storedPlan.id,
        courseCode: course.courseCode || null,
        courseName: course.courseName,
        normalizedName: normalizeName(course.courseName),
        cfu: course.cfu,
        studyYear: course.studyYear ?? null,
        semester: course.semester ?? null,
        courseType: course.courseType || 'mandatory',
        electiveGroup: course.electiveGroup || null,
        electiveRequiredCfu: course.electiveRequiredCfu ?? null,
        prerequisitesJson: JSON.stringify(course.prerequisites || [])
      });
    }

    return storedPlan;
  });

  return save();
}

function getStudyPlan(db, courseIdOrName, cohortYear) {
  if (!courseIdOrName || !cohortYear) return null;

  // 1. Cerca per codice esatto del corso
  let plan = db.prepare(`
    SELECT * FROM StudyPlans
    WHERE course_id = ? AND cohort_year = ?
    ORDER BY imported_at DESC, id DESC
    LIMIT 1
  `).get(courseIdOrName, cohortYear);

  if (plan) return plan;

  // 2. Fallback: cerca confrontando il nome normalizzato del corso
  const searchNormalized = normalizeName(courseIdOrName);
  return db.prepare(`
    SELECT * FROM StudyPlans
    WHERE LOWER(REPLACE(course_name, ' ', '')) LIKE '%' || REPLACE(?, ' ', '') || '%'
      AND cohort_year = ?
    ORDER BY imported_at DESC, id DESC
    LIMIT 1
  `).get(searchNormalized, cohortYear) || null;
}

function getPlanCourses(db, planId) {
  return db.prepare(`
    SELECT * FROM PlanCourses WHERE plan_id = ?
    ORDER BY study_year IS NULL, study_year, semester IS NULL, semester, course_name
  `).all(planId).map(parsePlanCourse);
}

    .all(profileId).map(row => row.plan_course_id);
}

function replaceSelectedElectives(db, profileId, planId, courseIds) {
  const uniqueIds = [...new Set(courseIds)];
  if (uniqueIds.length > 0) {
    const placeholders = uniqueIds.map(() => '?').join(', ');
    const valid = db.prepare(`
      SELECT COUNT(*) AS count FROM PlanCourses
      WHERE plan_id = ? AND course_type = 'elective' AND id IN (${placeholders})
    `).get(planId, ...uniqueIds).count;
    if (valid !== uniqueIds.length) {
      const error = new Error('Una o più scelte non appartengono al piano o non sono esami a scelta.');
      error.status = 400;
      throw error;
    }
  }

  const replace = db.transaction(() => {
    db.prepare('DELETE FROM SelectedElectives WHERE profile_id = ?').run(profileId);
    const insert = db.prepare('INSERT INTO SelectedElectives (profile_id, plan_course_id) VALUES (?, ?)');
    for (const courseId of uniqueIds) insert.run(profileId, courseId);
  });
  replace();
  return uniqueIds;
}

function clearSelectedElectives(db, profileId) {
  db.prepare('DELETE FROM SelectedElectives WHERE profile_id = ?').run(profileId);
}

const careerTaxesRoutes = require('./routes/career-taxes');

const authModule = require('./middleware/auth');
const authMiddleware = authModule.authMiddleware || authModule;
const { csrfProtection } = require('./lib/csrf');

const { db } = require('./db');
const { initCluster, closeCluster } = require('./scraper/pool');

const app = express();
const PORT = process.env.PORT || 3001;
const HOST = process.env.HOST || '0.0.0.0';

if (process.env.TRUST_PROXY) {
  app.set('trust proxy', Number(process.env.TRUST_PROXY) || 1);
}

// 🛡️ Sicurezza Intestazioni HTTP
app.use(helmet());

// 🌐 Configurazione CORS
const frontendOrigins = [...new Set([
  'http://localhost:3000',
  'http://127.0.0.1:3000',
  process.env.FRONTEND_URL
].filter(Boolean))];

app.use(cors({ origin: frontendOrigins, credentials: true }));

// 🔑 Protezione CSRF
app.use(csrfProtection);

// 🩺 Health Check (Senza Rate Limit)
app.get('/health', (_, res) => res.json({ status: 'ok', ts: new Date().toISOString() }));

// ⏱️ Rate Limiting Generale
app.use(rateLimit({ windowMs: 60 * 1000, max: 100, message: { error: 'Troppe richieste.' } }));

// 🔒 Rate Limiting Severo per i Login
const loginLimiter = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 10,
  standardHeaders: true,
  legacyHeaders: false,
  message: { error: 'Troppi tentativi di accesso. Riprova tra qualche minuto.' }
});
app.use(['/api/auth/login'], loginLimiter);

// 📦 Body Parser JSON
app.use(express.json({ limit: '200kb', strict: false }));
app.use((req, _res, next) => {
  if (req.body === undefined) req.body = {};
  next();
});

/* ------------------------------------------------------------------ */
/* 🔓 ROTTE PUBBLICHE                                                 */
/* ------------------------------------------------------------------ */
app.use('/api/session', sessionRoutes);
app.use('/api/auth', authRoutes);

/* ------------------------------------------------------------------ */

/* ------------------------------------------------------------------ */
app.use('/api', (req, res) => {
  res.status(404).json({ error: `Endpoint API non trovato: ${req.originalUrl}` });
});

app.use((err, req, res, next) => {
  console.error('[Server Error]', err);
  const status = err.status || 500;
  res.status(status).json({
    error: status >= 500 ? 'Errore interno del server.' : (err.message || 'Richiesta non valida.')
  });
});

/* ------------------------------------------------------------------ */
let server = null;
let shuttingDown = false;

async function start() {
  await initCluster();
  server = app.listen(PORT, HOST, () => {
    console.log(`[System] API attive su http://${HOST}:${PORT}`);
  });
}

async function gracefulShutdown(signal) {
  if (shuttingDown) return;
  shuttingDown = true;
  console.log(`\n[System] ${signal} ricevuto: spegnimento sicuro...`);

  const forceExit = setTimeout(() => {
    console.error('[System] Spegnimento forzato dopo 10s.');
    process.exit(1);
  }, 10000);
  forceExit.unref();

  try {
    await closeCluster();
  } catch (error) {
    console.error('[System] Errore chiusura cluster:', error.message);
  }

  try {
    db.close();
  } catch (error) {
    console.error('[System] Errore chiusura database:', error.message);
  }

  if (!server) {
    clearTimeout(forceExit);
    process.exit(0);
    return;
  }

  server.close(() => {
    clearTimeout(forceExit);
    console.log('[System] Server HTTP chiuso.');
    process.exit(0);
  });
}

process.on('SIGTERM', () => gracefulShutdown('SIGTERM'));
process.on('SIGINT', () => gracefulShutdown('SIGINT'));

start().catch(error => {
  console.error('[System] Fallimento critico avvio:', error);
  process.exit(1);
});

const { fetchAnnouncementsForProfessors } = require('./scraper/docenti');

// Uso:

const DEFAULT_PROFESSORS = [
  'DAVIDE MASSAROTTI',
  'VALERIA VITTORINI',
  'MARCO LOPS',
  'OFELIA PISANTI',
  'MATTIA BRESCIA'
];

const args = process.argv.slice(2);
const showAll = args.includes('--tutti');
const names = args.filter(a => !a.startsWith('--'));
const professors = names.length ? names : DEFAULT_PROFESSORS;
const LIMIT = showAll ? Infinity : 3;

(async () => {
  const start = Date.now();
  const groups = await fetchAnnouncementsForProfessors(professors);

  console.log(`\n===== AVVISI (${groups.length}/${professors.length} docenti trovati) =====`);

    console.log(`\n${'='.repeat(70)}\n${group.professor} — ${group.items.length} avvisi\nBacheca: ${group.boardUrl}\n${'='.repeat(70)}`);

      console.log('  ' + '-'.repeat(60));

      console.log('  ' + '-'.repeat(60));

      if (item.attachments.length) {

        item.attachments.forEach(a => console.log(`   - ${a.name}: ${a.url}`));

    }
  }

})().catch(err => {
  console.error('Errore nel test:', err);
  process.exit(1);
