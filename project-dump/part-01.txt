const fs = require('fs');
const path = require('path');

const outputFile = 'tutto_il_progetto.txt';
const excludeDirs = ['node_modules', '.next', '.git', 'dist', 'build', 'coverage'];
const includeExtensions = ['.js', '.ts', '.tsx', '.json', '.env'];

let outputContent = '';

function walkDir(dir) {
  const files = fs.readdirSync(dir);
  for (const file of files) {
    const fullPath = path.join(dir, file);
    const stat = fs.statSync(fullPath);

    if (stat.isDirectory()) {
      if (!excludeDirs.includes(file)) {
        walkDir(fullPath);
      }
    } else {
      const ext = path.extname(file);
      if (includeExtensions.includes(ext) || file === '.env') {
        const relativePath = path.relative(process.cwd(), fullPath);
        try {
          const content = fs.readFileSync(fullPath, 'utf8');
          outputContent += `\n\n==================================================\n`;
          outputContent += `FILE: ${relativePath}\n`;
          outputContent += `==================================================\n\n`;
          outputContent += content;
          console.log(`[OK] Aggiunto: ${relativePath}`);
        } catch (err) {
          console.log(`[SKIP] Impossibile leggere: ${relativePath}`);
        }
      }
    }
  }
}

console.log('Scansione e unione di tutti i file del progetto in corso...');
walkDir(process.cwd());

fs.writeFileSync(outputFile, outputContent, 'utf8');
console.log(`\nFatto! Il file cumulativo è pronto: ${path.resolve(outputFile)}`);

const checks = [];

function check(name, ok, message) {
  checks.push({ name, ok, message });
}

check('JWT_SECRET', Boolean(process.env.JWT_SECRET && process.env.JWT_SECRET.trim()),
  isProduction ? 'obbligatorio in produzione' : 'mancante: verrà usato il segreto di sviluppo');

const username = process.env.DOCENTI_USERNAME || '';
const password = process.env.DOCENTI_PASSWORD || '';
const hasDocentiCredentials = Boolean(username && password);
check('DOCENTI_USERNAME', !username || username === username.trim(), 'presenta spazi ai bordi');
check('DOCENTI_PASSWORD', !password || password === password.trim(), 'presenta spazi ai bordi');

check('PORT', !process.env.PORT || /^\d+$/.test(process.env.PORT), 'deve essere un numero');
check('FRONTEND_URL', !process.env.FRONTEND_URL || /^https?:\/\//.test(process.env.FRONTEND_URL), 'deve essere un URL http(s)');

for (const { name, ok, message } of checks) {
  console.log(`${ok ? '[OK]' : '[!]'} ${name}: ${ok ? 'valida' : message}`);
}

console.log(`\nDocenti: ${hasDocentiCredentials ? 'credenziali presenti' : 'credenziali assenti (solo avvisi pubblici)'}`);
console.log(`Ambiente: ${process.env.NODE_ENV || 'development'}`);

const Database = require('better-sqlite3');
const path = require('path');

const db = new Database(path.join(__dirname, 'app.db'));
db.pragma('journal_mode = WAL');

db.exec(`
  CREATE TABLE IF NOT EXISTS StudentProfile (
    id TEXT PRIMARY KEY,
    course_id TEXT,
    course_name TEXT,
    cohort_year INTEGER,
    current_course_year INTEGER,
    channel TEXT,
    surname_initial TEXT,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
  );

  CREATE TABLE IF NOT EXISTS StudyPlans (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    course_id TEXT NOT NULL,
    cohort_year INTEGER NOT NULL,
    course_name TEXT NOT NULL,
    version TEXT DEFAULT '1',
    source_name TEXT,
    source_url TEXT,
    imported_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(course_id, cohort_year, version)
  );

  CREATE TABLE IF NOT EXISTS PlanCourses (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    plan_id INTEGER NOT NULL,
    course_code TEXT,
    course_name TEXT NOT NULL,
    normalized_name TEXT NOT NULL,
    cfu INTEGER NOT NULL,
    study_year INTEGER,
    semester INTEGER,
    course_type TEXT DEFAULT 'mandatory',
    elective_group TEXT,
    elective_required_cfu INTEGER,
    prerequisites_json TEXT DEFAULT '[]',
    FOREIGN KEY (plan_id) REFERENCES StudyPlans(id) ON DELETE CASCADE
  );

  CREATE TABLE IF NOT EXISTS SelectedElectives (
    profile_id TEXT NOT NULL,
    plan_course_id INTEGER NOT NULL,
    PRIMARY KEY (profile_id, plan_course_id),
    FOREIGN KEY (profile_id) REFERENCES StudentProfile(id) ON DELETE CASCADE,
    FOREIGN KEY (plan_course_id) REFERENCES PlanCourses(id) ON DELETE CASCADE
  );

  CREATE TABLE IF NOT EXISTS UserSessions (
    token TEXT PRIMARY KEY,
    profile_id TEXT NOT NULL,
    created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    expires_at DATETIME NOT NULL,
    FOREIGN KEY (profile_id) REFERENCES StudentProfile(id) ON DELETE CASCADE
  );
`);

module.exports = { db };

function normalizeName(value) {
  return String(value || '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, ' ')
    .trim();
}

function parsePlanCourse(row) {
  if (!row) return null;
  let prerequisites = [];
  try {
    prerequisites = JSON.parse(row.prerequisites_json || '[]');
  } catch {}
  return {
    ...row,
    courseCode: row.course_code,
    courseName: row.course_name,
    studyYear: row.study_year,
    courseType: row.course_type,
    electiveGroup: row.elective_group,
    electiveRequiredCfu: row.elective_required_cfu,
    prerequisites
  };
}

function getStudentProfile(db, profileId) {
  return db.prepare('SELECT * FROM StudentProfile WHERE id = ?').get(profileId) || null;
}

function upsertStudentProfile(db, profile) {
  db.prepare(`
    INSERT INTO StudentProfile (id, course_id, course_name, cohort_year, current_course_year, channel, surname_initial)
    VALUES (@id, @courseId, @courseName, @cohortYear, @currentCourseYear, @channel, @surnameInitial)
    ON CONFLICT(id) DO UPDATE SET
      course_id = COALESCE(excluded.course_id, StudentProfile.course_id),
      course_name = COALESCE(excluded.course_name, StudentProfile.course_name),
      cohort_year = COALESCE(excluded.cohort_year, StudentProfile.cohort_year),
      current_course_year = COALESCE(excluded.current_course_year, StudentProfile.current_course_year),
      channel = COALESCE(excluded.channel, StudentProfile.channel),
      surname_initial = COALESCE(excluded.surname_initial, StudentProfile.surname_initial),
      updated_at = CURRENT_TIMESTAMP
  `).run({
    id: profile.id,
    courseId: profile.courseId || null,
    courseName: profile.courseName || null,
    cohortYear: profile.cohortYear || null,
    currentCourseYear: profile.currentCourseYear || null,
    channel: profile.channel || null,
    surnameInitial: profile.surnameInitial || null
  });
  return getStudentProfile(db, profile.id);
}

function updateStudentProfile(db, profileId, updates) {
  const columns = {
    courseId: 'course_id',
    courseName: 'course_name',
    cohortYear: 'cohort_year',
    currentCourseYear: 'current_course_year',
    channel: 'channel',
    surnameInitial: 'surname_initial'
  };
  const entries = Object.entries(columns).filter(([key]) => Object.hasOwn(updates, key));
  if (entries.length === 0) return getStudentProfile(db, profileId);

  const assignments = entries.map(([, column]) => `${column} = @${column}`).join(', ');
  const values = Object.fromEntries(entries.map(([key, column]) => [column, updates[key] ?? null]));
  db.prepare(`UPDATE StudentProfile SET ${assignments}, updated_at = CURRENT_TIMESTAMP WHERE id = @id`)
    .run({ ...values, id: profileId });
  return getStudentProfile(db, profileId);
}

function saveStudyPlan(db, plan, courses) {
  const save = db.transaction(() => {
    db.prepare(`
      INSERT INTO StudyPlans (course_id, cohort_year, course_name, version, source_name, source_url)
      VALUES (@courseId, @cohortYear, @courseName, @version, @sourceName, @sourceUrl)
      ON CONFLICT(course_id, cohort_year, version) DO UPDATE SET
        course_name = excluded.course_name,
        source_name = excluded.source_name,
        source_url = excluded.source_url,
        imported_at = CURRENT_TIMESTAMP
    `).run({
      courseId: plan.courseId,
      cohortYear: plan.cohortYear,
      courseName: plan.courseName,
      version: String(plan.version || '1'),
      sourceName: plan.sourceName || null,
      sourceUrl: plan.sourceUrl || null
    });

    const storedPlan = db.prepare(`
      SELECT * FROM StudyPlans
      WHERE course_id = ? AND cohort_year = ? AND version = ?
    `).get(plan.courseId, plan.cohortYear, String(plan.version || '1'));

    db.prepare('DELETE FROM PlanCourses WHERE plan_id = ?').run(storedPlan.id);
    const insertCourse = db.prepare(`
      INSERT INTO PlanCourses (
        plan_id, course_code, course_name, normalized_name, cfu, study_year, semester,
        course_type, elective_group, elective_required_cfu, prerequisites_json
      ) VALUES (
        @planId, @courseCode, @courseName, @normalizedName, @cfu, @studyYear, @semester,
        @courseType, @electiveGroup, @electiveRequiredCfu, @prerequisitesJson
      )
    `);

    for (const course of courses) {
      insertCourse.run({
        planId: storedPlan.id,
        courseCode: course.courseCode || null,
        courseName: course.courseName,
        normalizedName: normalizeName(course.courseName),
        cfu: course.cfu,
        studyYear: course.studyYear ?? null,
        semester: course.semester ?? null,
        courseType: course.courseType || 'mandatory',
        electiveGroup: course.electiveGroup || null,
        electiveRequiredCfu: course.electiveRequiredCfu ?? null,
        prerequisitesJson: JSON.stringify(course.prerequisites || [])
      });
    }

    return storedPlan;
  });

  return save();
}

function getStudyPlan(db, courseIdOrName, cohortYear) {
  if (!courseIdOrName || !cohortYear) return null;

  // 1. Cerca per codice esatto del corso
  let plan = db.prepare(`
    SELECT * FROM StudyPlans
    WHERE course_id = ? AND cohort_year = ?
    ORDER BY imported_at DESC, id DESC
    LIMIT 1
  `).get(courseIdOrName, cohortYear);

  if (plan) return plan;

  // 2. Fallback: cerca confrontando il nome normalizzato del corso
  const searchNormalized = normalizeName(courseIdOrName);
  return db.prepare(`
    SELECT * FROM StudyPlans
    WHERE LOWER(REPLACE(course_name, ' ', '')) LIKE '%' || REPLACE(?, ' ', '') || '%'
      AND cohort_year = ?
    ORDER BY imported_at DESC, id DESC
    LIMIT 1
  `).get(searchNormalized, cohortYear) || null;
}

function getPlanCourses(db, planId) {
  return db.prepare(`
    SELECT * FROM PlanCourses WHERE plan_id = ?
    ORDER BY study_year IS NULL, study_year, semester IS NULL, semester, course_name
  `).all(planId).map(parsePlanCourse);
}

function getSelectedElectiveIds(db, profileId) {
  return db.prepare('SELECT plan_course_id FROM SelectedElectives WHERE profile_id = ?')
    .all(profileId).map(row => row.plan_course_id);
}

function replaceSelectedElectives(db, profileId, planId, courseIds) {
  const uniqueIds = [...new Set(courseIds)];
  if (uniqueIds.length > 0) {
    const placeholders = uniqueIds.map(() => '?').join(', ');
    const valid = db.prepare(`
      SELECT COUNT(*) AS count FROM PlanCourses
      WHERE plan_id = ? AND course_type = 'elective' AND id IN (${placeholders})
    `).get(planId, ...uniqueIds).count;
    if (valid !== uniqueIds.length) {
      const error = new Error('Una o più scelte non appartengono al piano o non sono esami a scelta.');
      error.status = 400;
      throw error;
    }
  }

  const replace = db.transaction(() => {
    db.prepare('DELETE FROM SelectedElectives WHERE profile_id = ?').run(profileId);
    const insert = db.prepare('INSERT INTO SelectedElectives (profile_id, plan_course_id) VALUES (?, ?)');
    for (const courseId of uniqueIds) insert.run(profileId, courseId);
  });
  replace();
  return uniqueIds;
}

function clearSelectedElectives(db, profileId) {
  db.prepare('DELETE FROM SelectedElectives WHERE profile_id = ?').run(profileId);
}

module.exports = {
  normalizeName,
  getStudentProfile,
  upsertStudentProfile,
  updateStudentProfile,
  saveStudyPlan,
  getStudyPlan,
  getPlanCourses,
  getSelectedElectiveIds,
  replaceSelectedElectives,
  clearSelectedElectives
};

const path = require('path');
const fs = require('fs');

const SCHEMA_VERSION = 1;
const DEFAULT_DB_PATH = path.resolve(__dirname, '../data/unicorno.sqlite');

function ensureDataDir() {
  const dataDir = path.dirname(DEFAULT_DB_PATH);
  if (!fs.existsSync(dataDir)) fs.mkdirSync(dataDir, { recursive: true });
}

function initSchema(database) {
  database.exec(`
    -- Tabella Sessioni Utente (autenticazione opzionale)
    CREATE TABLE IF NOT EXISTS UserSessions (
      token TEXT PRIMARY KEY,
      profile_id TEXT NOT NULL,
      expires_at DATETIME NOT NULL
    );

    CREATE INDEX IF NOT EXISTS idx_user_sessions_expires
      ON UserSessions (expires_at);

    -- Profilo studente (anagrafica e configurazione carriera)
    CREATE TABLE IF NOT EXISTS StudentProfile (
      id TEXT PRIMARY KEY,
      course_id TEXT,
      course_name TEXT,
      cohort_year INTEGER,
      current_course_year INTEGER,
      channel TEXT,
      surname_initial TEXT,
      carriera_json TEXT,
      tasse_json TEXT,
      updated_at DATETIME DEFAULT CURRENT_TIMESTAMP
    );

    -- Manifesti degli studi
    CREATE TABLE IF NOT EXISTS StudyPlans (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      course_id TEXT NOT NULL,
      cohort_year INTEGER NOT NULL,
      course_name TEXT,
      version TEXT DEFAULT '1',
      source_name TEXT,
      source_url TEXT,
      imported_at DATETIME DEFAULT CURRENT_TIMESTAMP,
      UNIQUE(course_id, cohort_year, version)
    );

    -- Insegnamenti del piano di studi
    CREATE TABLE IF NOT EXISTS PlanCourses (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      plan_id INTEGER NOT NULL,
      course_code TEXT,
      course_name TEXT NOT NULL,
      normalized_name TEXT,
      cfu INTEGER,
      study_year INTEGER,
      semester INTEGER,
      course_type TEXT DEFAULT 'mandatory',
      elective_group TEXT,
      elective_required_cfu INTEGER,
      prerequisites_json TEXT,
      FOREIGN KEY (plan_id) REFERENCES StudyPlans(id) ON DELETE CASCADE
    );

    -- Esami a scelta selezionati
    CREATE TABLE IF NOT EXISTS SelectedElectives (
      profile_id TEXT NOT NULL,
      plan_course_id INTEGER NOT NULL,
      PRIMARY KEY (profile_id, plan_course_id),
      FOREIGN KEY (profile_id) REFERENCES StudentProfile(id) ON DELETE CASCADE,
      FOREIGN KEY (plan_course_id) REFERENCES PlanCourses(id) ON DELETE CASCADE
    );
  `);
}

/**
 * Crea un'istanza isolata del database (usare ':memory:' nei test).
 */
function createDatabase(filename = DEFAULT_DB_PATH) {
  if (filename !== ':memory:') ensureDataDir();
  const database = new Database(filename);
  database.pragma('journal_mode = WAL');
  database.pragma('foreign_keys = ON');
  initSchema(database);
  database.pragma(`user_version = ${SCHEMA_VERSION}`);
  return database;
}

const dbIndex = require('./index');
const db = dbIndex.db || dbIndex;

// 1. Utility di base
function normalizeName(value) {
  return String(value || '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, ' ')
    .trim();
}

function parsePlanCourse(row) {
  if (!row) return null;
  let prerequisites = [];
  try {
    prerequisites = JSON.parse(row.prerequisites_json || '[]');
  } catch {}
  
  return {
    ...row,
    courseCode: row.course_code,
    courseName: row.course_name,
    studyYear: row.study_year,
    courseType: row.course_type,
    electiveGroup: row.elective_group,
    electiveRequiredCfu: row.elective_required_cfu,
    prerequisites
  };
}

// 2. Risolutore DB e Parametri (DRY)
function resolveDb(possibleDb) {
  if (possibleDb?.prepare) return possibleDb;
  if (possibleDb?.db?.prepare) return possibleDb.db;
  if (dbIndex?.prepare) return dbIndex;
  if (dbIndex?.db?.prepare) return dbIndex.db;
  throw new Error("Istanza Database SQLite non trovata.");
}

/**
 * Estrae l'istanza DB e il parametro reale, supportando la doppia firma:
 * func(db, arg) oppure func(arg)
 */
function getDbAndArg(arg1, arg2) {
  try {
    const database = resolveDb(arg1);
    return { db: database, arg: database === arg1 ? arg2 : arg1 };
  } catch {
    return { db: resolveDb(), arg: arg1 };
  }
}

function getStudentProfile(arg1, arg2) {
  const { db, arg: profileId } = getDbAndArg(arg1, arg2);
  return db.prepare('SELECT * FROM StudentProfile WHERE id = ?').get(profileId) || null;
}

function upsertStudentProfile(arg1, arg2) {
  const { db, arg: profile } = getDbAndArg(arg1, arg2);
  const targetId = profile.id || profile.profileId || profile.codiceFiscale;

  db.prepare(`
    INSERT INTO StudentProfile (
      id, course_id, course_name, cohort_year, current_course_year, channel, surname_initial,
      carriera_json, tasse_json
    )
    VALUES (@id, @courseId, @courseName, @cohortYear, @currentCourseYear, @channel, @surnameInitial, @carrieraJson, @tasseJson)
    ON CONFLICT(id) DO UPDATE SET
      course_id = COALESCE(excluded.course_id, StudentProfile.course_id),
      course_name = COALESCE(excluded.course_name, StudentProfile.course_name),
      cohort_year = COALESCE(excluded.cohort_year, StudentProfile.cohort_year),
      current_course_year = COALESCE(excluded.current_course_year, StudentProfile.current_course_year),
      channel = COALESCE(excluded.channel, StudentProfile.channel),
      surname_initial = COALESCE(excluded.surname_initial, StudentProfile.surname_initial),
      carriera_json = COALESCE(excluded.carriera_json, StudentProfile.carriera_json),
      tasse_json = COALESCE(excluded.tasse_json, StudentProfile.tasse_json),
      updated_at = CURRENT_TIMESTAMP
  `).run({
    id: targetId,
    courseId: profile.courseId || profile.anagrafica?.codiceCorso || null,
    courseName: profile.courseName || profile.anagrafica?.corsoDiLaurea || null,
    cohortYear: profile.cohortYear || profile.anagrafica?.annoImmatricolazione || null,
    currentCourseYear: profile.currentCourseYear || profile.anagrafica?.annoCorso || null,
    channel: profile.channel || null,
    surnameInitial: profile.surnameInitial || null,
    carrieraJson: profile.esamiSostenuti ? JSON.stringify(profile.esamiSostenuti) : null,
    tasseJson: profile.tasse ? JSON.stringify(profile.tasse) : null
  });

  return getStudentProfile(db, targetId);
}

function updateStudentProfile(arg1, profileIdOrUpdates, optionalUpdates) {
  const { db, arg: profileId } = getDbAndArg(arg1, profileIdOrUpdates);
  const updates = (db === arg1) ? optionalUpdates : profileIdOrUpdates;

  const columns = {
    courseId: 'course_id',
    courseName: 'course_name',
    cohortYear: 'cohort_year',
    currentCourseYear: 'current_course_year',
    channel: 'channel',
    surnameInitial: 'surname_initial'
  };
  
  const entries = Object.entries(columns).filter(([key]) => Object.hasOwn(updates || {}, key));
  if (entries.length === 0) return getStudentProfile(db, profileId);

  const assignments = entries.map(([, column]) => `${column} = @${column}`).join(', ');
  const values = Object.fromEntries(entries.map(([key, column]) => [column, updates[key] ?? null]));
  
  db.prepare(`UPDATE StudentProfile SET ${assignments}, updated_at = CURRENT_TIMESTAMP WHERE id = @id`)
    .run({ ...values, id: profileId });

  return getStudentProfile(db, profileId);
}

function saveStudyPlan(arg1, planOrCourses, optionalCourses) {
  const { db, arg: plan } = getDbAndArg(arg1, planOrCourses);
  const courses = (db === arg1) ? optionalCourses : planOrCourses;

  return db.transaction(() => {
    db.prepare(`
      INSERT INTO StudyPlans (course_id, cohort_year, course_name, version, source_name, source_url)
      VALUES (@courseId, @cohortYear, @courseName, @version, @sourceName, @sourceUrl)
      ON CONFLICT(course_id, cohort_year, version) DO UPDATE SET
        course_name = excluded.course_name,
        source_name = excluded.source_name,
        source_url = excluded.source_url,
        imported_at = CURRENT_TIMESTAMP
    `).run({
      courseId: plan.courseId,
      cohortYear: plan.cohortYear,
      courseName: plan.courseName,
      version: String(plan.version || '1'),
      sourceName: plan.sourceName || null,
      sourceUrl: plan.sourceUrl || null
    });

    const storedPlan = db.prepare(`
      SELECT * FROM StudyPlans WHERE course_id = ? AND cohort_year = ? AND version = ?
    `).get(plan.courseId, plan.cohortYear, String(plan.version || '1'));

    db.prepare('DELETE FROM PlanCourses WHERE plan_id = ?').run(storedPlan.id);
    
    const insertCourse = db.prepare(`
      INSERT INTO PlanCourses (
        plan_id, course_code, course_name, normalized_name, cfu, study_year, semester,
        course_type, elective_group, elective_required_cfu, prerequisites_json
      ) VALUES (
        @planId, @courseCode, @courseName, @normalizedName, @cfu, @studyYear, @semester,
        @courseType, @electiveGroup, @electiveRequiredCfu, @prerequisitesJson
      )
    `);

    for (const course of (courses || [])) {
      insertCourse.run({
        planId: storedPlan.id,
        courseCode: course.courseCode || null,
        courseName: course.courseName,
        normalizedName: normalizeName(course.courseName),
        cfu: course.cfu,
        studyYear: course.studyYear ?? null,
        semester: course.semester ?? null,
        courseType: course.courseType || 'mandatory',
        electiveGroup: course.electiveGroup || null,
        electiveRequiredCfu: course.electiveRequiredCfu ?? null,
        prerequisitesJson: JSON.stringify(course.prerequisites || [])
      });
    }

    return storedPlan;
  })();
}

function getStudyPlan(arg1, courseOrCohort, optionalCohort) {
  const { db, arg: courseIdOrName } = getDbAndArg(arg1, courseOrCohort);
  const cohortYear = (db === arg1) ? optionalCohort : courseOrCohort;

  if (!courseIdOrName || !cohortYear) return null;

  let plan = db.prepare(`
    SELECT * FROM StudyPlans WHERE course_id = ? AND cohort_year = ?
    ORDER BY imported_at DESC, id DESC LIMIT 1
  `).get(courseIdOrName, cohortYear);

  if (plan) return plan;

  return db.prepare(`
    SELECT * FROM StudyPlans
    WHERE LOWER(REPLACE(course_name, ' ', '')) LIKE '%' || REPLACE(?, ' ', '') || '%'
      AND cohort_year = ?
    ORDER BY imported_at DESC, id DESC LIMIT 1
  `).get(normalizeName(courseIdOrName), cohortYear) || null;
}

function getPlanCourses(arg1, arg2) {
  const { db, arg: planId } = getDbAndArg(arg1, arg2);
  return db.prepare(`
    SELECT * FROM PlanCourses WHERE plan_id = ?
    ORDER BY study_year IS NULL, study_year, semester IS NULL, semester, course_name
  `).all(planId).map(parsePlanCourse);
}

function getSelectedElectiveIds(arg1, arg2) {
  const { db, arg: profileId } = getDbAndArg(arg1, arg2);
  return db.prepare('SELECT plan_course_id FROM SelectedElectives WHERE profile_id = ?')
    .all(profileId).map(row => row.plan_course_id);
}

function replaceSelectedElectives(arg1, profileIdOrPlanId, planIdOrCourseIds, optionalCourseIds) {
  const { db, arg: profileId } = getDbAndArg(arg1, profileIdOrPlanId);
  const planId = (db === arg1) ? planIdOrCourseIds : profileIdOrPlanId;
  const courseIds = (db === arg1) ? optionalCourseIds : planIdOrCourseIds;

  const uniqueIds = [...new Set(courseIds || [])];
  
  if (uniqueIds.length > 0) {
    const placeholders = uniqueIds.map(() => '?').join(', ');
    const valid = db.prepare(`
      SELECT COUNT(*) AS count FROM PlanCourses
      WHERE plan_id = ? AND course_type = 'elective' AND id IN (${placeholders})
    `).get(planId, ...uniqueIds).count;

    if (valid !== uniqueIds.length) {
      const error = new Error('Una o più scelte non appartengono al piano o non sono esami a scelta.');
      error.status = 400;
      throw error;
    }
  }

  db.transaction(() => {
    db.prepare('DELETE FROM SelectedElectives WHERE profile_id = ?').run(profileId);
    const insert = db.prepare('INSERT INTO SelectedElectives (profile_id, plan_course_id) VALUES (?, ?)');
    for (const courseId of uniqueIds) insert.run(profileId, courseId);
  })();

  return uniqueIds;
}

function clearSelectedElectives(arg1, arg2) {
  const { db, arg: profileId } = getDbAndArg(arg1, arg2);
  db.prepare('DELETE FROM SelectedElectives WHERE profile_id = ?').run(profileId);
}

function academicYearIndex(cohort?: number | string | null) {
  const start = Number(cohort);
  if (!start) return null;
  const now = new Date();
  const currentStart = now.getMonth() >= 8 ? now.getFullYear() : now.getFullYear() - 1;
  const idx = currentStart - start + 1;
  return idx >= 1 ? idx : null;
}

function formatEuro(val?: number | null) {
  if (val === undefined || val === null) return '0,00 €';
  return new Intl.NumberFormat('it-IT', { style: 'currency', currency: 'EUR' }).format(val);
}

function formatDate(dateStr?: string | null) {
  if (!dateStr) return '—';
  if (/^\d{4}-\d{2}-\d{2}$/.test(dateStr)) {
    const [y, m, d] = dateStr.split('-');
    return `${d}/${m}/${y}`;
  }
  return dateStr;
}

// Converte una data dd/mm/yyyy (o yyyy-mm-dd) in timestamp; 0 se non riconosciuta.
function dataItalianaTs(value?: string | null): number {
  if (!value) return 0;
  const testo = String(value);
  const it = testo.match(/(\d{1,2})\/(\d{1,2})\/(\d{4})/);
  if (it) return new Date(Number(it[3]), Number(it[2]) - 1, Number(it[1])).getTime();
  const iso = testo.match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (iso) return new Date(Number(iso[1]), Number(iso[2]) - 1, Number(iso[3])).getTime();
  return 0;
}

function isAppelloPassato(p: Prenotazione): boolean {
  const t = dataItalianaTs(p.dataAppello);
  if (!t) return false;
  const oggi = new Date();
  oggi.setHours(0, 0, 0, 0);
  return t < oggi.getTime();
}

// Date leggibili anche quando arrivano come timestamp in millisecondi.
function dataLeggibile(value?: string | null): string {
  if (!value) return '—';
  const testo = String(value).trim();
  if (/^\d{12,14}$/.test(testo)) {
    const data = new Date(Number(testo));
    if (!Number.isNaN(data.getTime())) {
      return data.toLocaleDateString('it-IT', { day: '2-digit', month: '2-digit', year: 'numeric' });
    }
  }
  return formatDate(testo);
}

  const [data, setData] = useState<Metriche | null>(null);
  const [loading, setLoading] = useState(false);
  const [loginStep, setLoginStep] = useState('');
  const [error, setError] = useState('');
  const [onboardingRequired, setOnboardingRequired] = useState(false);
  const [editingProfile, setEditingProfile] = useState(false);
  const [activeView, setActiveView] = useState<View>('home');

  const [searchExam, setSearchExam] = useState('');
  const [searchPrenotazione, setSearchPrenotazione] = useState('');
  const [taxTab, setTaxTab] = useState<'pending' | 'paid'>('pending');
  const [slowNotice, setSlowNotice] = useState<'loading' | 'ready' | 'error' | null>(null);

  const clearSession = () => {
    setToken(null);
    setProfile(null);
    setData(null);
    setOnboardingRequired(false);
    setEditingProfile(false);
    setActiveView('home');
    setCredentials({ codiceFiscale: '', password: '', email: '' });
  };

  const caricaDati = async (jwtToken: string) => {
    await Promise.resolve();
    setLoading(true);
    setError('');
    try {
      const metriche = await fetchMetriche(jwtToken);
      setData(metriche);
    } catch (err: unknown) {
      if (err instanceof Error && err.message === 'SESSION_EXPIRED') {
        clearSession();
        setError('Sessione scaduta. Inserisci di nuovo le credenziali.');
      } else {
        setError(err instanceof Error ? err.message : 'Impossibile sincronizzare i dati.');
      }
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    let active = true;

    restoreSession()
      .then(async session => {
        if (!active || !session) return;
        setToken(session.token);
        setProfile(session.profile);
        setOnboardingRequired(session.onboardingRequired);
        await caricaDati(session.token);
      })
      .finally(() => {
        if (active) setCheckingSession(false);
      });

    return () => { active = false; };
  }, []);

  useEffect(() => {
    if (!token) {
      Promise.resolve().then(() => setSlowNotice(null));
      return;
    }

    let active = true;
    let timer: ReturnType<typeof setTimeout> | null = null;

    const poll = async () => {
      try {
        const slow: SlowData = await fetchSlowData(token);
        if (!active) return;

        if (slow.status === 'ready') {
          setData(prev => (prev ? {
            ...prev,
            tasse: slow.taxes ?? prev.tasse,
            prenotazioni: slow.prenotazioni ?? prev.prenotazioni
          } : prev));
          setSlowNotice('ready');
          timer = setTimeout(() => { if (active) setSlowNotice(null); }, 5000);
          return;
        }

        if (slow.status === 'error') {
          setSlowNotice('error');
          return;
        }

        setSlowNotice('loading');
        timer = setTimeout(poll, 3000);
      } catch (error) {
        if (!active) return;
        if (error instanceof Error && error.message === 'SESSION_EXPIRED') return;
        setSlowNotice('error');
        timer = setTimeout(poll, 10000);
      }
    };

    Promise.resolve().then(() => {
      if (!active) return;
      setSlowNotice('loading');
      return poll();
    });

    return () => {
      active = false;
      if (timer) clearTimeout(timer);
    };
  }, [token]);

const handleLogin = async (e: React.FormEvent) => {
    e.preventDefault();

    const emailIstituzionale = credentials.email?.trim() || '';

    // 1. Controllo che email e password siano inserite
    if (!emailIstituzionale || !credentials.password) {
      setError('Inserisci email istituzionale e password.');
      return;
    }

    // 2. Controllo che l'email abbia un formato valido
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(emailIstituzionale)) {
      setError('Inserisci una email istituzionale valida (es. nome.cognome@studenti.unina.it).');
      return;
    }

    setLoading(true);
    setError('');
    setLoginStep('Connessione ai servizi Unina...');
    
    const steps = [
      'Connessione a Segrepass...',
      'Lettura carriera e profilo...',
      'Estrazione esami e metriche...',
      'Preparazione dashboard...',
      'Avvio acquisizione dati Docenti...'
    ];
    let stepIndex = 0;
    const stepTimer = setInterval(() => {
      stepIndex = Math.min(stepIndex + 1, steps.length - 1);
      setLoginStep(steps[stepIndex]);
    }, 2200);

    try {
      // 3. IL TRUCCO: Inviamo l'email sia come codice fiscale (Segrepass lo accetta!) sia come email per i Docenti
      const access = await loginAPI({
        codiceFiscale: emailIstituzionale, 
        password: credentials.password,
        email: emailIstituzionale
      });
      
      setLoginStep('Finalizzazione sessione...');
      setToken(access.token);
      setProfile(access.profile);
      setOnboardingRequired(access.onboardingRequired);
      setData(access.metriche);
      setCredentials({ codiceFiscale: '', password: '', email: '' });
    } catch (err: unknown) {
      setError(err instanceof Error ? err.message : 'Accesso non riuscito. Controlla le credenziali.');
    } finally {
      clearInterval(stepTimer);
      setLoginStep('');
      setLoading(false);
    }
  };
  const handleLogout = async () => {
    await logoutAPI();
    clearSession();
    setError('');
  };

  const handleProfileSaved = (savedProfile: ProfiloStudente) => {
    setProfile(savedProfile);
    setOnboardingRequired(false);
    setEditingProfile(false);
  };

  const extProfile: ProfiloRecord = profile ?? {};
  const anagrafica = data?.anagrafica;

  const matricola = anagrafica?.matricola || extProfile?.matricola || null;
  const nomeCompleto = anagrafica?.nomeCompleto || extProfile?.name || 'Studente';
  const corsoDiLaurea = anagrafica?.corsoDiLaurea || extProfile?.courseName || extProfile?.course_name || 'Corso non specificato';
  const codiceCorso = anagrafica?.codiceCorso || extProfile?.courseId || extProfile?.course_id || null;
  const annoCorso = anagrafica?.annoCorso || academicYearIndex(extProfile?.cohortYear || extProfile?.cohort_year);
  const coorte = anagrafica?.annoImmatricolazione || extProfile?.cohortYear || extProfile?.cohort_year || '—';
  const canale = extProfile?.channel || 'Da impostare';

  const pendingTaxes: TassaItem[] = taxesObj?.pending || taxesObj?.inDebito || [];

  const totalPaid = taxesObj?.summary?.totalPaid ?? paidTaxes.reduce((sum, item) => sum + (item.amount ?? item.importo ?? 0), 0);
  const totalPending = taxesObj?.summary?.totalPending ?? pendingTaxes.reduce((sum, item) => sum + (item.amount ?? item.importo ?? 0), 0);
  const nextDeadline = taxesObj?.summary?.nextDeadline || pendingTaxes.map(t => t.dueDate || t.scadenza).filter(Boolean).sort()[0];

  const [docentiBookings, setDocentiBookings] = useState<BookedExamDocenti[]>([]);

 const fetchingDocentiRef = useRef(false);

  useEffect(() => {

    if (fetchingDocentiRef.current) return; // Blocca le chiamate parallele

    fetchingDocentiRef.current = true; // Segnala che la chiamata è in corso

         // Aspetta 2 secondi prima di sbloccare, evitando il rimbalzo di React Strict Mode
         setTimeout(() => { fetchingDocentiRef.current = false; }, 2000); 
      });

    return () => { active = false; };

  const listaPrenotazioni = useMemo<Prenotazione[]>(() => {

      if (visti.has(chiave)) return false;
      visti.add(chiave);
      return true;

      if (passataA !== passataB) return passataA ? 1 : -1;
      const tA = dataItalianaTs(a.dataAppello);
      const tB = dataItalianaTs(b.dataAppello);
      return passataA ? tB - tA : tA - tB;

  const prenotazioniFiltrate = useMemo(() => {

    return listaPrenotazioni.filter(p => {
      const mat = (p.materia || '').toLowerCase();

      return mat.includes(query) || doc.includes(query);

  // Appelli Segrepass futuri: azione principale. Tutto il resto (passato o già prenotato) è storico.
  const appelliDisponibili = useMemo(
    () => prenotazioniFiltrate.filter(p => !isAppelloPassato(p) && !p.id?.startsWith('docenti-')),
    [prenotazioniFiltrate]
  );
  const storicoPrenotazioni = useMemo(
    () => prenotazioniFiltrate.filter(p => !appelliDisponibili.includes(p)),
    [prenotazioniFiltrate, appelliDisponibili]

    'min-h-screen bg-[radial-gradient(ellipse_at_top,#0B3A6B_0%,#061426_45%,#030912_100%)] text-slate-100';

  if (!token) {
    return (
      <main className={`${pageBg} flex items-center justify-center p-4`}>
        <form onSubmit={handleLogin} className={`${CARD} w-full max-w-sm space-y-4 p-6`}>
          <div className="space-y-1 text-center">
            <div className="mx-auto mb-3 grid h-14 w-14 place-items-center rounded-full border-2 border-[#E6C887]/70 text-sm font-bold text-[#E6C887]">
              FII
            </div>
            <h1 className="text-2xl font-bold tracking-tight text-[#E6C887]">Federico II · Studente</h1>
            <p className="text-sm text-slate-400">Accedi con la tua email istituzionale</p>
          </div>
          {error && <p role="alert" className="text-center text-xs font-medium text-red-400">{error}</p>}
          {loading && loginStep && (
            <p className="text-center text-xs font-medium text-[#5AB0F0]">{loginStep}</p>
          )}

          <input
            type="email"
            name="emailIstituzionale"
            autoComplete="email"
            placeholder="Email istituzionale (@studenti.unina.it)"
            value={credentials.email ?? ''}
            onChange={(e) => setCredentials((c) => ({ ...c, email: e.target.value }))}
            className="w-full rounded-xl border border-white/10 bg-white/5 py-2.5 px-3 text-xs text-white placeholder-slate-400 focus:border-[#5AB0F0] focus:outline-none"
          />
          <input

            autoComplete="current-password"

          <button
            type="submit"
            disabled={loading}
            className="flex min-h-12 w-full items-center justify-center gap-2 rounded-xl bg-[#2F7FD0] px-4 font-semibold text-white transition hover:bg-[#3D8FE2] disabled:opacity-50"
          >
            <LogIn className="h-4 w-4" />
            {loading ? 'Accesso in corso...' : 'Accedi'}
          </button>
        </form>
      </main>
    );
  }




  if (profile && editingProfile) {
    return (
      <main className={`${pageBg} px-4 py-8 sm:px-8`}>
        <ProfileSetupView

          initialProfile={profile}

  return (
    <div className={pageBg}>
      {slowNotice && (
        <div
          role="status"
          aria-live="polite"
          className={`fixed right-4 top-4 z-50 max-w-xs rounded-xl border px-4 py-3 text-xs shadow-lg backdrop-blur ${
            slowNotice === 'loading'
              ? 'border-[#3B7BC4]/50 bg-[#0A1B31]/95 text-slate-200'
              : slowNotice === 'ready'

              : 'border-amber-600/60 bg-amber-950/95 text-amber-100'

        >
          {slowNotice === 'loading' && 'Tasse e appelli in preparazione: i dati già disponibili sono visibili subito.'}

          >
            FII

                  selected

function readableError(err: unknown): string {
  const message = err instanceof Error ? err.message : '';
  if (message === 'SESSION_EXPIRED') return 'Sessione scaduta. Esci e accedi di nuovo.';
  if (/CREDENTIALS_REQUIRED/i.test(message)) {
    return 'Credenziali Docenti non disponibili nella sessione. Esci e accedi di nuovo con l\u2019email istituzionale.';
  }
  if (/non valide/i.test(message)) return 'Credenziali Docenti non valide.';
  if (/Nessun dato Docenti/i.test(message)) return 'Nessun dato Docenti trovato sul portale. Riprova più tardi.';
  return message || 'Errore durante il caricamento dei dati Docenti.';
}

function formatDate(dateStr?: string | null) {
  if (!dateStr) return '—';
  try {
    const d = new Date(dateStr);
    if (isNaN(d.getTime())) return dateStr;
    return d.toLocaleDateString('it-IT', {
      day: '2-digit',
      month: '2-digit',
      year: 'numeric',
      hour: '2-digit',
      minute: '2-digit'
    });
  } catch {
    return dateStr;
  }
}

function getTimestamp(dateStr?: string | null) {
  if (!dateStr) return 0;
  const t = new Date(dateStr).getTime();
  return isNaN(t) ? 0 : t;
}

function stripTags(html?: string | null) {
  if (!html) return '';
  return html.replace(/<[^>]*>/g, '').trim();
}

import { useEffect, useState, useCallback } from 'react';
import { Sun, Moon, Monitor } from 'lucide-react';
import { getSavedTheme, saveTheme, resolveTheme, applyTheme, type Theme } from '@/lib/theme';

export default function ThemeSwitch() {
  const [theme, setTheme] = useState<Theme>('system');
  const [resolved, setResolved] = useState<'light' | 'dark'>('light');

  const apply = useCallback((t: Theme) => {
    setTheme(t);
    saveTheme(t);
    const r = resolveTheme(t);
    setResolved(r);
    applyTheme(r);
  }, []);

  useEffect(() => {
    apply(getSavedTheme());

    if (theme === 'system') {
      const mq = window.matchMedia('(prefers-color-scheme: dark)');
      const handler = (e: MediaQueryListEvent) => {
        setResolved(e.matches ? 'dark' : 'light');
        applyTheme(e.matches ? 'dark' : 'light');
      };
      mq.addEventListener('change', handler);
      return () => mq.removeEventListener('change', handler);
    }
  }, [theme]);

  const labels: Record<Theme, string> = { light: 'Chiaro', dark: 'Scuro', system: 'Sistema' };
  const icons: Record<Theme, React.ReactNode> = {
    light: <Sun className="h-5 w-5" />,
    dark: <Moon className="h-5 w-5" />,
    system: <Monitor className="h-5 w-5" />,
  };

  const next: Theme = theme === 'light' ? 'dark' : theme === 'dark' ? 'system' : 'light';

  return (
    <button
      onClick={() => apply(next)}
      aria-label={`Tema: ${labels[theme]} (${resolved})`}
      title={`Tema: ${labels[theme]} → ${resolved === 'dark' ? 'scuro' : 'chiaro'}`}
      className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg border border-white/10 bg-white/5 text-slate-200 transition hover:bg-white/10 focus-visible:outline focus-visible:outline-2 focus-visible:outline-[#5AB0F0] group"
    >
      {icons[theme]}
      <span className="absolute -right-8 top-1/2 -translate-y-1/2 z-50 whitespace-nowrap text-[10px] font-medium text-slate-400 bg-zinc-900/90 px-2 py-1 rounded border border-white/10 opacity-0 group-hover:opacity-100 transition-opacity">
        {labels[theme]} ({resolved === 'dark' ? 'scuro' : 'chiaro'})
      </span>
      <span className="sr-only">Cambia tema: {labels[theme]}</span>
    </button>

function authHeaders(token?: string): Record<string, string> {
  const headers: Record<string, string> = { 'Content-Type': 'application/json' };
  if (token) headers.Authorization = `Bearer ${token}`;
  return headers;
}

async function fetchWithAuth(url: string, token: string, options: RequestInit = {}) {
  const res = await fetch(url, {
    ...options,
    credentials: 'include',
    headers: {
      ...authHeaders(token),
      ...(options.headers || {})
    }
  });

  if (res.status === 401 || res.status === 403) {
    throw new Error('SESSION_EXPIRED');
  }

  const data = await res.json().catch(() => ({}));
  if (!res.ok) {
    throw new Error(data.error || 'Errore di comunicazione con il server.');
  }
  return data;
}

export async function loginAPI(credentials: CredenzialiSegrepass): Promise<LoginResponse> {
  const res = await fetch(apiUrl('/api/auth/login'), {
    method: 'POST',
    credentials: 'include',
    headers: authHeaders(),
    body: JSON.stringify(credentials)
  });

  const data = await res.json().catch(() => ({}));

  if (!res.ok) {
    throw new Error(data.error || 'Credenziali non valide o login fallito.');
  }

  return data as LoginResponse;
}

export async function restoreSession(): Promise<RestoreSessionResponse | null> {
  const res = await fetch(apiUrl('/api/auth/session'), {
    method: 'GET',
    credentials: 'include',
    headers: authHeaders()
  });
  if (!res.ok) return null;
  return (await res.json()) as RestoreSessionResponse;
}

  if (typeof window === 'undefined') return 'system';
  const v = localStorage.getItem(KEY);
  if (v === 'light' || v === 'dark' || v === 'system') return v;
  return 'system';
}

export function saveTheme(t: Theme) {
  if (typeof window === 'undefined') return;
  localStorage.setItem(KEY, t);
}

export function resolveTheme(t: Theme): 'light' | 'dark' {
  if (t === 'system') {
    return typeof window !== 'undefined' && window.matchMedia('(prefers-color-scheme: dark)').matches
      ? 'dark'
      : 'light';
  }
  return t;
}

export function applyTheme(t: 'light' | 'dark') {
  if (typeof document === 'undefined') return;
  const root = document.documentElement;
  if (t === 'dark') root.classList.add('dark');
  else root.classList.remove('dark');
}

async function extractCourses(page) {
  return await page.evaluate(() => {
    const courses = [];
    const cardElements = document.querySelectorAll('.card, article, [class*="card"], div.bg-white');

    cardElements.forEach(card => {
      const text = card.innerText || '';
      const codeMatch = text.match(/\(([A-Z0-9]{3,5})\)/);

      if (!codeMatch) return;

      const code = codeMatch[1];
      const lines = text
        .split('\n')
        .map(line => line.trim())
        .filter(Boolean);

      const nameLine = lines.find(line => line.includes(`(${code})`));
      if (!nameLine) return;

      const name = nameLine.replace(`(${code})`, '').trim();
      if (!name) return;

      const department = lines.find(line => 
        line !== nameLine && !line.includes(`(${code})`)
      ) || 'DIPARTIMENTO GENERALE';

      courses.push({
        code,
        name,
        department
      });
    });

    return courses;
  });
}

    try {
      console.log(`📄 Pagina ${pageNum}/${CONFIG.endPage} - Tentativo ${attempt}`);

      const response = await page.goto(url, {
        waitUntil: 'networkidle2',
        timeout: CONFIG.timeout
      });

      if (!response || !response.ok()) {
        throw new Error(`HTTP ${response ? response.status() : 'nessuna risposta'}`);
      }

      await page.waitForFunction(
        () => document.body && document.body.innerText.length > 0,
        { timeout: 10000 }
      );

      const courses = await extractCourses(page);
      return courses;
    } catch (error) {
      console.error(`⚠️ Errore pagina ${pageNum}: ${error.message}`);

      if (attempt === CONFIG.retries) {
        console.error(`❌ Pagina ${pageNum} non elaborata.`);
        return null;
      }

      await sleep(1000 * attempt);
    }
  }

  return null;
}

const COOKIE_NAME = 'segrepass_token';
const COOKIE_MAX_AGE_MS = 7 * 24 * 60 * 60 * 1000;

function parseCookies(header = '') {
  const entries = String(header)
    .split(';')
    .map(part => {
      const separator = part.indexOf('=');
      if (separator === -1) return null;
      const name = part.slice(0, separator).trim();
      if (!name) return null;
      try {
        return [name, decodeURIComponent(part.slice(separator + 1).trim())];
      } catch {
        return [name, part.slice(separator + 1).trim()];
      }
    })
    .filter(Boolean);

  return Object.fromEntries(entries);
}

function cookieOptions() {
  return {
    httpOnly: true,
    sameSite: 'lax',
    secure: process.env.NODE_ENV === 'production',
    path: '/',
    maxAge: COOKIE_MAX_AGE_MS
  };
}

function setAuthCookie(res, token) {
  res.cookie(COOKIE_NAME, token, cookieOptions());
}

function clearAuthCookie(res) {
  const { maxAge, ...options } = cookieOptions();
  res.clearCookie(COOKIE_NAME, options);
}

function readAuthToken(req) {
  return parseCookies(req.headers.cookie)[COOKIE_NAME] || null;
}

module.exports = { COOKIE_NAME, setAuthCookie, clearAuthCookie, readAuthToken, parseCookies };

const SAFE_METHODS = new Set(['GET', 'HEAD', 'OPTIONS']);
const DEFAULT_ORIGINS = ['http://localhost:3000', 'http://127.0.0.1:3000'];

function normalizeOrigin(value) {
  if (!value) return null;
  try {
    const url = new URL(String(value));
    if (url.protocol !== 'http:' && url.protocol !== 'https:') return null;
    return url.origin;
  } catch {
    return null;
  }
}

function configuredOrigins(env = process.env) {
  return String(env.FRONTEND_URL || '')
    .split(',')
    .map(value => normalizeOrigin(value.trim()))
    .filter(Boolean);
}

function allowedOrigins(req, env = process.env) {
  const origins = new Set([...DEFAULT_ORIGINS, ...configuredOrigins(env)]);

  // X-Forwarded-Host è attendibile solo se il proxy è dichiarato esplicitamente.
  const trustProxy = env.TRUST_PROXY && !['0', 'false', 'off'].includes(String(env.TRUST_PROXY).toLowerCase());
  const forwardedHost = trustProxy ? req.headers['x-forwarded-host'] : null;
  const host = forwardedHost || req.headers.host;
  const protocol = String(req.protocol || 'http').split(',')[0].trim();
  const sameOrigin = normalizeOrigin(`${protocol}://${host}`);
  if (sameOrigin) origins.add(sameOrigin);

  return origins;
}

function requestOrigin(req) {
  // Origin è la fonte affidabile; Referer è il fallback per i browser che lo omettono.
  return normalizeOrigin(req.headers.origin) || normalizeOrigin(req.headers.referer);
}

/**
 * Protezione CSRF per le richieste mutanti autenticate tramite cookie.
 * - I metodi safe non sono interessati.
 * - Le richieste con Bearer esplicito sono immuni al CSRF (l'header non è
 *   impostabile da una pagina di terze parti senza preflight CORS).
 * - Tutte le altre richieste mutanti devono avere Origin o Referer consentiti.
 */
function csrfProtection(req, res, next) {
  if (SAFE_METHODS.has(req.method)) return next();

  const authorization = req.headers.authorization || '';
  if (/^Bearer\s+\S+/i.test(authorization)) return next();

  const origin = requestOrigin(req);
  if (!origin) {
    return res.status(403).json({
      code: 'CSRF_ORIGIN_REQUIRED',
      error: 'Richiesta bloccata: manca l’intestazione Origin o Referer.'
    });
  }

  if (!allowedOrigins(req).has(origin)) {
    return res.status(403).json({
      code: 'CSRF_ORIGIN_INVALID',
      error: 'Origine della richiesta non consentita.'
    });
  }

  return next();
}

module.exports = {
  csrfProtection,
  normalizeOrigin,
  allowedOrigins,
  requestOrigin,
  configuredOrigins,
  DEFAULT_ORIGINS,
  SAFE_METHODS
};

/**
 * Modulo per il matching tra corsi del piano di studi ed esami sostenuti
 */

const DEFAULT_THRESHOLD = 0.84;
const AMBIGUITY_MARGIN = 0.06;

const ROMAN_NUMERALS = [
  [/\biv\b/gi, '4'],
  [/\biii\b/gi, '3'],
  [/\bii\b/gi, '2'],
  [/\bi\b/gi, '1'],
  [/\bed\b/gi, 'e']
];

function normalizeCourseName(value) {
  let str = String(value || '')
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .toLowerCase();

  for (const [pattern, replacement] of ROMAN_NUMERALS) {
    str = str.replace(pattern, replacement);
  }

  return str.replace(/[^a-z0-9]+/g, ' ').trim();
}

function levenshteinDistance(left, right) {
  let previous = Array.from({ length: right.length + 1 }, (_, index) => index);
  for (let leftIndex = 1; leftIndex <= left.length; leftIndex++) {
    const current = [leftIndex];
    for (let rightIndex = 1; rightIndex <= right.length; rightIndex++) {
      const substitution = previous[rightIndex - 1] + (left[leftIndex - 1] === right[rightIndex - 1] ? 0 : 1);
      current[rightIndex] = Math.min(previous[rightIndex] + 1, current[rightIndex - 1] + 1, substitution);
    }
    previous = current;
  }
  return previous[right.length];
}

function nameSimilarity(left, right) {
  const normalizedLeft = normalizeCourseName(left);
  const normalizedRight = normalizeCourseName(right);
  if (!normalizedLeft || !normalizedRight) return 0;
  if (normalizedLeft === normalizedRight) return 1;

  const longest = Math.max(normalizedLeft.length, normalizedRight.length);
  const editSimilarity = 1 - levenshteinDistance(normalizedLeft, normalizedRight) / longest;

  const leftTokens = new Set(normalizedLeft.split(' '));
  const rightTokens = new Set(normalizedRight.split(' '));
  const intersection = [...leftTokens].filter(token => rightTokens.has(token)).length;
  const union = new Set([...leftTokens, ...rightTokens]).size;
  const tokenSimilarity = union ? intersection / union : 0;

  return Math.max(editSimilarity, tokenSimilarity);
}

function getCourseCode(course) {
  return String(course.courseCode || course.codice || course.cod || '').trim().toLowerCase();
}

function getCourseName(course) {
  return course.courseName || course.materia || course.insegnamento || course.name || '';
}

function scoreCandidate(planCourse, passedExam) {
  const planCode = getCourseCode(planCourse);
  const passedCode = getCourseCode(passedExam);

  if (planCode && passedCode) {
    if (planCode !== passedCode) return null;
    const creditsMatch = !planCourse.cfu || !passedExam.cfu || Math.abs(planCourse.cfu - passedExam.cfu) < 0.5;
    return { score: creditsMatch ? 1 : 0.82, method: 'course_code', needsReview: !creditsMatch };
  }

  const similarity = nameSimilarity(getCourseName(planCourse), getCourseName(passedExam));
  if (similarity < DEFAULT_THRESHOLD) return null;

  const creditsMatch = !planCourse.cfu || !passedExam.cfu || Math.abs(planCourse.cfu - passedExam.cfu) < 0.5;
  return {
    score: Math.max(0, Math.min(1, similarity + (creditsMatch ? 0.04 : -0.12))),
    method: 'fuzzy_name',
    needsReview: !creditsMatch
  };
}

function matchStudyPlan({ planCourses = [], passedExams = [], selectedElectiveIds = [] }) {
  const usedExamIndexes = new Set();
  // Gli ID possono essere numerici (manifesto) o codici stringa (piano da Segrepass).
  const selectedIds = new Set(selectedElectiveIds.map(String));

  const courses = planCourses.map(planCourse => {
    const candidates = passedExams
      .map((passedExam, index) => ({ passedExam, index, match: scoreCandidate(planCourse, passedExam) }))
      .filter(candidate => candidate.match && !usedExamIndexes.has(candidate.index))
      .sort((left, right) => right.match.score - left.match.score);

    const best = candidates[0];
    const isAmbiguous = best && candidates[1] && (best.match.score - candidates[1].match.score < AMBIGUITY_MARGIN);
    const isSelected = selectedIds.has(String(planCourse.id));
    const isUnselectedElective = planCourse.courseType === 'elective' && !isSelected;

    if (best && (isAmbiguous || best.match.needsReview)) {
      usedExamIndexes.add(best.index);
      return { ...planCourse, status: 'needs_review', confidence: best.match.score, matchMethod: best.match.method };
    }

    if (best && best.match.score >= DEFAULT_THRESHOLD) {
      usedExamIndexes.add(best.index);
      return {
        ...planCourse,
        status: 'passed',
        confidence: best.match.score,
        matchMethod: best.match.method,
        passedExam: best.passedExam
      };
    }

    return {
      ...planCourse,
      status: isUnselectedElective ? 'elective_available' : 'missing',
      selected: isSelected
    };
  });

  const electiveGroups = new Map();
  for (const course of courses.filter(item => item.courseType === 'elective')) {
    const groupId = course.electiveGroup || 'electives';
    const group = electiveGroups.get(groupId) || {
      id: groupId,
      requiredCfu: 0,
      passedCfu: 0,
      selectedPendingCfu: 0,
      courses: []
    };
    group.requiredCfu = Math.max(group.requiredCfu, Number(course.electiveRequiredCfu) || 0);
    if (course.status === 'passed') group.passedCfu += Number(course.cfu) || 0;
    else if (course.selected) group.selectedPendingCfu += Number(course.cfu) || 0;
    group.courses.push(course);
    electiveGroups.set(groupId, group);
  }

  const electiveSummary = [...electiveGroups.values()].map(group => ({
    ...group,
    remainingCfu: Math.max(0, group.requiredCfu - group.passedCfu)
  }));

  return {
    courses,
    electiveGroups: electiveSummary,
    summary: {
      passedCourses: courses.filter(course => course.status === 'passed').length,
      missingCourses: courses.filter(course => course.status === 'missing').length,
      reviewCourses: courses.filter(course => course.status === 'needs_review').length,
      mandatoryMissingCfu: courses
        .filter(course => course.courseType === 'mandatory' && course.status === 'missing')
        .reduce((total, course) => total + (Number(course.cfu) || 0), 0),
      electiveRemainingCfu: electiveSummary.reduce((total, group) => total + group.remainingCfu, 0)
    }
  };
}

/**

const CFU_TOTALI_CORSO = parseInt(process.env.CFU_TOTALI_CORSO || '180', 10);

function normalizzaIntestazione(value) {
  return String(value || '').normalize('NFD').replace(/[\u0300-\u036f]/g, '').toLowerCase().trim();
}

function normalizzaEsame(r) {
  const getVal = (patterns) => {
    const keys = Object.keys(r || {});
    for (const pat of patterns) {
      const match = keys.find(k => pat.test(normalizzaIntestazione(k)));
      if (match && r[match] !== null && r[match] !== undefined && String(r[match]).trim()) {
        return String(r[match]).trim();
      }
    }
    return '';
  };

  const votoRaw = getVal([/^valutazione/, /^voto/, /^giudizio/, /^esito/]);
  const lode = /30\s*l|lode/i.test(votoRaw);
  const votoMatch = votoRaw.match(/\d+(?:[.,]\d+)?/);
  const votoNum = votoMatch ? parseFloat(votoMatch[0].replace(',', '.')) : NaN;
  const cfuRaw = getVal([/cfu/, /^crediti/, /^peso/]);
  const cfu = parseInt(cfuRaw, 10) || 0;

  const materia = getVal([/^insegnamento/, /^denominazione/, /^descrizione/, /^materia/, /^esame/, /^attivita/, /^nome/]) || 'Esame';
  const dataEsame = getVal([/^data/, /^verbalizz/, /^superamento/]);

  return {
    codice: getVal([/^codice/, /^cod/]) || null,
    materia,
    voto: isNaN(votoNum) || votoNum === 0 ? null : votoNum,
    giudizio: isNaN(votoNum) || votoNum === 0 ? (votoRaw || 'IDONEO') : null,
    lode,
    cfu,
    data: dataEsame || null
  };
}

function calcolaMetriche(records = [], cfuTotali = CFU_TOTALI_CORSO) {
  if (!Array.isArray(records) || records.length === 0) {
    return {
      esami: [],
      cfuAcquisiti: 0,
      cfuTotali,
      avanzamentoCFU: 0,
      mediaAritmetica: '--',
      mediaPonderata: '--',
      baseLaurea: '--'
    };
  }

  const tuttiEsami = records.map(normalizzaEsame).filter(e => e.materia && (e.cfu > 0 || e.voto !== null || e.giudizio));
  const esamiConVoto = tuttiEsami.filter(e => typeof e.voto === 'number' && e.voto > 0 && e.cfu > 0);

  const cfuAcquisiti = tuttiEsami.reduce((acc, e) => acc + (e.cfu || 0), 0);
  const cfuPonderati = esamiConVoto.reduce((acc, e) => acc + (e.cfu || 0), 0);

  let mediaAritmetica = null;
  let mediaPonderata = null;
  let baseLaurea = null;

  if (esamiConVoto.length > 0) {
    mediaAritmetica = esamiConVoto.reduce((acc, e) => acc + e.voto, 0) / esamiConVoto.length;
    mediaPonderata = esamiConVoto.reduce((acc, e) => acc + (e.voto * e.cfu), 0) / (cfuPonderati || 1);
    baseLaurea = (mediaPonderata / 30) * 110;
  }

  const avanzamentoCFU = Math.min(Math.round((cfuAcquisiti / cfuTotali) * 100), 100);

  return {
    esami: tuttiEsami,
    cfuAcquisiti,
    cfuTotali,
    avanzamentoCFU,
    mediaAritmetica: mediaAritmetica ? parseFloat(mediaAritmetica.toFixed(2)) : '--',
    mediaPonderata: mediaPonderata ? parseFloat(mediaPonderata.toFixed(2)) : '--',
    baseLaurea: baseLaurea ? parseFloat(baseLaurea.toFixed(2)) : '--'
  };
}

function simulaBaseLaurea(esamiAttuali = [], esamiFuturi = []) {
  const tutti = [...esamiAttuali, ...esamiFuturi].filter(e => typeof e.voto === 'number' && e.cfu > 0);
  if (!tutti.length) return 0;
  const totCFU = tutti.reduce((acc, e) => acc + e.cfu, 0);
  const mp = tutti.reduce((acc, e) => acc + (e.voto * e.cfu), 0) / totCFU;
  return parseFloat(((mp / 30) * 110).toFixed(2));
}

// lib/piano-normalizer.js
// Converte la tabella "Insegnamenti del Piano" letta da Segrepass nel formato canonico
// usato dal matching. Le colonne vengono riconosciute dal nome dell'intestazione,
// non dalla posizione, così eventuali cambi di ordine non rompono lo scraper.

const COLUMN_RULES = [
  ['code', /^codice$/i],
  ['name', /descrizione/i],
  ['year', /^anno/i],
  ['outcome', /^esito$/i],
  ['grade', /^voto$/i],
  ['cfu', /^cfu$/i],
  ['sector', /^settore$/i]
];

const YEAR_BY_WORD = { PRIMO: 1, SECONDO: 2, TERZO: 3, QUARTO: 4, QUINTO: 5, SESTO: 6 };
const FREE_CHOICE = /\bA\s+SCELTA\b/i;

function clean(value) {
  return String(value ?? '').replace(/\s+/g, ' ').trim();
}

function mapColumns(headers = []) {
  const map = {};
  headers.map(clean).forEach((header, index) => {
    for (const [key, rule] of COLUMN_RULES) {
      if (map[key] === undefined && rule.test(header)) {
        map[key] = index;
        break;
      }
    }
  });
  return map;
}

/**
 * @param {{headers: string[], rows: string[][]}} table
 * @returns {Array<object>} insegnamenti nel formato canonico (vedi README in fondo al file)
 */
function normalizePianoTable(table) {
  if (!table || !Array.isArray(table.rows)) return [];

  const col = mapColumns(table.headers);
  if (col.code === undefined || col.name === undefined || col.cfu === undefined) return [];

  const get = (cells, key) => (col[key] === undefined ? '' : clean(cells[col[key]]));
  const seen = new Set();
  const courses = [];

  for (const cells of table.rows) {
    const code = get(cells, 'code');
    const name = get(cells, 'name');
    if (!code || !name || seen.has(code)) continue;
    seen.add(code);

    const yearText = get(cells, 'year').toUpperCase();
    const grade = parseInt(get(cells, 'grade'), 10);
    const sector = get(cells, 'sector');
    const freeChoice = FREE_CHOICE.test(name);

    courses.push({
      id: code,
      code,
      courseCode: code,
      name,
      courseName: name,
      cfu: parseInt(get(cells, 'cfu'), 10) || 0,
      year: YEAR_BY_WORD[yearText] ?? (parseInt(yearText, 10) || null),
      semester: null,
      // Euristica: gli insegnamenti "A SCELTA ..." sono a scelta libera.
      courseType: freeChoice ? 'elective' : 'mandatory',
      electiveGroup: freeChoice ? 'free_choice' : null,
      electiveRequiredCfu: null,
      sector: sector && sector !== '0' ? sector : null,
      // Esito e voto sono dello studente che ha letto la pagina: vanno usati solo per visualizzazione.
      outcome: get(cells, 'outcome') || null,
      grade: Number.isFinite(grade) ? grade : null
    });
  }

  return courses;
}

function stripHtml(raw) {
  if (!raw) return '';
  return String(raw)
    .replace(/<[^>]*>/g, '')
    .replace(/&nbsp;/g, ' ')
    .replace(/&amp;/g, '&')
    .replace(/&lt;/g, '<')
    .replace(/&gt;/g, '>')
    .replace(/\s+/g, ' ')
    .trim();
}

function cleanProfessorName(raw) {
  if (!raw) return 'Non specificato';
  let cleaned = stripHtml(raw)
    .replace(/^(prof\.?|dott\.?|ing\.?|presidente|commissione|docente[:\s]*)/gi, '')
    .replace(/\s+/g, ' ')
    .trim();
  return cleaned || 'Non specificato';
}

function parseItalianDate(dateStr) {
  if (!dateStr) return 0;
  const match = String(dateStr).match(/(\d{1,2})\/(\d{1,2})\/(\d{4})/);
  if (!match) return 0;
  return new Date(Number(match[3]), Number(match[2]) - 1, Number(match[1])).getTime();
}

function normalizePrenotazioni(input = [], subscriptions = [], studyPlanCourses = []) {
  const itemsToProcess = [];

  if (Array.isArray(input)) {
    for (const item of input) {
      if (item && item.materia) {
        itemsToProcess.push(item);
      } else if (item && item.records) {
        for (const rec of item.records) {
          const materia = rec.insegnamento || rec.materia || rec.descrizione;
          if (materia) {
            itemsToProcess.push({
              materia,
              docente: rec.docente || rec.professore || rec.commissione || 'Non specificato',
              dataAppello: rec.data || rec.appello,
              ora: rec.ora,
              aula: rec.aula,
              note: rec.note
            });
          }
        }
      }
    }
  }

  const rawList = [];
  for (const item of itemsToProcess) {
    const materia = stripHtml(item.materia);
    if (!materia || materia.length < 2) continue;

    const docente = cleanProfessorName(item.docente);
    const dataAppello = stripHtml(item.dataAppello || item.data || 'Da definire');
    const ora = stripHtml(item.ora);
    const aula = stripHtml(item.aula);
    const note = stripHtml(item.note);

    const matchingSub = (subscriptions || []).find(sub => {
      if (!sub.professor) return false;
      const p1 = docente.toLowerCase();
      const p2 = stripHtml(sub.professor).toLowerCase();
      return p1.includes(p2) || p2.includes(p1);
    });

    rawList.push({
      id: item.id || `pren_${Math.random().toString(36).substr(2, 9)}`,
      materia,
      docente: matchingSub?.professor || docente,
      dataAppello,
      ora,
      aula,
      note,
      stato: 'in_attesa',
      isMyProfessor: Boolean(matchingSub),
      matchedCourse: matchingSub ? matchingSub.courseName : null
    });
  }

  // Deduplicazione
  const uniqueList = [];
  const seen = new Set();
  for (const item of rawList) {
    const key = `${item.materia.toLowerCase()}_${item.docente.toLowerCase()}_${item.dataAppello}`;
    if (!seen.has(key)) {
      seen.add(key);
      uniqueList.push(item);
    }
  }

  // Ordinamento: Prima "Il tuo professore", poi per data appello crescente
  return uniqueList.sort((a, b) => {
    if (b.isMyProfessor !== a.isMyProfessor) {
      return Number(b.isMyProfessor) - Number(a.isMyProfessor);
    }
    return parseItalianDate(a.dataAppello) - parseItalianDate(b.dataAppello);
  });
}

// lib/secret-box.js
// Cifratura simmetrica (AES-256-GCM) per tenere la password dentro il JWT
// senza renderla leggibile a chi possiede il token.
const crypto = require('crypto');
const { JWT_SECRET } = require('./config');

function key() {
  // CRED_KEY è consigliata (chiave dedicata); in sua assenza si deriva da JWT_SECRET.
  const material = process.env.CRED_KEY || `cred:${JWT_SECRET}`;
  return crypto.createHash('sha256').update(material).digest();
}

function seal(text) {
  const iv = crypto.randomBytes(12);
  const cipher = crypto.createCipheriv('aes-256-gcm', key(), iv);
  const enc = Buffer.concat([cipher.update(String(text), 'utf8'), cipher.final()]);
  return [iv, cipher.getAuthTag(), enc].map(b => b.toString('base64')).join('.');
}

function open(sealed) {
  try {
    const [iv, tag, enc] = String(sealed).split('.').map(part => Buffer.from(part, 'base64'));
    const decipher = crypto.createDecipheriv('aes-256-gcm', key(), iv);
    decipher.setAuthTag(tag);
    return Buffer.concat([decipher.update(enc), decipher.final()]).toString('utf8');
  } catch {
    return null;
  }
}

module.exports = { seal, open };

function parseAmount(raw) {
  if (typeof raw === 'number') return Number.isFinite(raw) ? raw : 0;
  const text = String(raw ?? '').replace(/[€\s]/g, '');
  if (!text) return 0;

  let normalized = text;
  if (text.includes(',') && text.includes('.')) {
    normalized = text.replace(/\./g, '').replace(',', '.');
  } else if (text.includes(',')) {
    normalized = text.replace(',', '.');
  }

  const value = Number.parseFloat(normalized);
  return Number.isFinite(value) ? value : 0;
}

function parseDate(raw) {
  const text = String(raw ?? '').trim();
  if (!text) return null;

  const iso = text.match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (iso) return `${iso[1]}-${iso[2]}-${iso[3]}`;

  const italian = text.match(/^(\d{1,2})[\/.](\d{1,2})[\/.](\d{4})$/);
  if (italian) {
    return `${italian[3]}-${italian[2].padStart(2, '0')}-${italian[1].padStart(2, '0')}`;
  }
  return null;
}

function roundAmount(value) {
  return Math.round((Number(value) || 0) * 100) / 100;
}

function sumAmounts(items) {
  return roundAmount(items.reduce((total, item) => total + (Number(item?.importo) || 0), 0));
}

function findPaymentTables(tables) {
  const result = {
    paid: [],
    pending: [],
    summary: {
      totalPaid: 0,
      totalPending: 0,
      nextDeadline: null
    }
  };

  if (!Array.isArray(tables)) return result;

  tables.forEach(table => {
    const headers = (table.headers || []).map(header => String(header).toLowerCase());
    const records = table.records || [];

    const isDebitoTable = headers.some(header =>
      header.includes('codice versamento') ||
      header.includes('descrizione versamento') ||
      header.includes('paga') ||
      header.includes('importo')
    );

    const isPagatiTable = headers.some(header =>
      header.includes('data versamento') ||
      header.includes('data pagamento') ||
      header.includes('ricevuta') ||
      header.includes('quietanza')
    );

    records.forEach(record => {
      const keys = Object.keys(record);
      const getVal = term => {
        const key = keys.find(candidate => candidate.toLowerCase().includes(term));
        return key ? record[key] : '';
      };

      const rawImporto = getVal('importo') || '0';
      const descrizione = getVal('descrizione') || getVal('causale') || 'Contributo Universitario';
      const anno = getVal('anno') || '';
      const numImporto = parseAmount(rawImporto);
      const scadenzaRaw = getVal('scadenza');

      if (numImporto > 0 || descrizione) {
        const item = {
          codice: getVal('codice') || '7001',
          descrizione: anno ? `${descrizione} (${anno})` : descrizione,
          annoAccademico: anno,
          importo: numImporto,
          importoFormatted: rawImporto.includes('€') ? rawImporto : `€${numImporto.toFixed(2).replace('.', ',')}`,
          scadenza: scadenzaRaw || 'In sospeso',
          scadenzaISO: parseDate(scadenzaRaw),
          stato: isDebitoTable && !isPagatiTable ? 'DA_PAGARE' : 'PAGATO'
        };

        if (isPagatiTable) {
          result.paid.push(item);
        } else if (isDebitoTable) {
          result.pending.push(item);
        }
      }
    });
  });

  result.summary.totalPaid = sumAmounts(result.paid);
  result.summary.totalPending = sumAmounts(result.pending);
  result.summary.nextDeadline = result.pending
    .map(item => item.scadenzaISO)
    .filter(Boolean)
    .sort()[0] || null;

  return result;
}

/**
 * Normalizza lo snapshot tasse nel formato atteso dalle rotte e dal frontend.
 * Accetta sia { paid, pending } sia { pagati, inDebito }.

function buildTaxesSummary(taxes = {}) {
  const paid = Array.isArray(taxes.paid) ? taxes.paid : (Array.isArray(taxes.pagati) ? taxes.pagati : []);
  const pending = Array.isArray(taxes.pending) ? taxes.pending : (Array.isArray(taxes.inDebito) ? taxes.inDebito : []);
  const sourceSummary = taxes.summary || {};

  return {
    paid,
    pending,
    summary: {
      totalPaid: roundAmount(sourceSummary.totalPaid ?? sumAmounts(paid)),
      totalPending: roundAmount(sourceSummary.totalPending ?? sumAmounts(pending)),
      nextDeadline: sourceSummary.nextDeadline ?? null
    }
  };
}

// middleware/auth.js
const jwt = require('jsonwebtoken');
const { JWT_SECRET } = require('../lib/config');
const { readAuthToken } = require('../lib/cookies');
const { open } = require('../lib/secret-box');

function extractToken(req) {
  const header = req.headers.authorization || '';
  const match = header.match(/^Bearer\s+(.+)$/i);
  if (match) return match[1].trim();
  // Il token persistente vive in un cookie httpOnly, non in localStorage.
  return readAuthToken(req) || req.headers['x-session-token'] || null;
}

function authMiddleware(req, res, next) {
  const token = extractToken(req);
  if (!token) {
    return res.status(401).json({ error: 'Accesso negato: token di sessione mancante.' });
  }

  try {
    const payload = jwt.verify(token, JWT_SECRET);
    const profileId = payload.sub || payload.codiceFiscale;
    if (!profileId) {
      return res.status(401).json({ error: 'Sessione non valida: profilo assente.' });
    }

    req.user = {
      sub: payload.sub || payload.codiceFiscale,
      codiceFiscale: payload.codiceFiscale || payload.sub,
      email: payload.email || undefined,
      // La password viaggia cifrata nel JWT (campo "pw"): qui viene decifrata.
      password: payload.pw ? open(payload.pw) || undefined : undefined
    };
    return next();
  } catch (error) {
    const expired = error.name === 'TokenExpiredError';
    return res.status(401).json({
      error: expired ? 'Sessione scaduta. Effettua nuovamente il login.' : 'Sessione non valida.'
    });
  }
}

module.exports = { authMiddleware };

      return res.status(409).json({ code: 'CAREER_SYNC_REQUIRED', error: 'Accedi di nuovo a Segrepass per aggiornare gli esami.' });
    }

    const resolved = resolveStudyPlan(db, profile, req.user.sub);
    if (!resolved) {
      // Il frontend riconosce "non disponibile" e mostra il messaggio di corso non supportato.
      return res.status(404).json({
        code: 'STUDY_PLAN_NOT_AVAILABLE',
        error: 'Piano di studi non disponibile per questo corso.'
      });
    }

    const result = matchStudyPlan({
      planCourses: resolved.courses,
      passedExams,
      selectedElectiveIds: getSelectedElectiveIds(db, req.user.sub)
    });

    return res.json({ profile, plan: resolved.plan, source: resolved.source, ...result });
  } catch (error) {
    console.error('[Career Plan Error]:', error.message);
    return res.status(500).json({ error: 'Errore interno nel recupero del piano di studi.' });
  }
});

module.exports = router;

const express = require('express');
const router = express.Router();
const { getDatiCarriera } = require('../scraper/segrepass');

router.post('/', async (req, res) => {
  const { username, codiceFiscale, password } = req.body;
  const user = codiceFiscale || username;

  if (!user || !password) {
    return res.status(400).json({ success: false, error: 'CREDENTIALS_REQUIRED' });
  }

  try {
    const fullData = await getDatiCarriera(
      { codiceFiscale: user, password },
      { phase: 'slow' }
    );
    res.json({ success: true, data: fullData.tasse });
  } catch (error) {
    res.status(500).json({
      success: false,
      error: error.code || 'TAXES_ERROR',
      message: error.message
    });
  }
});

module.exports = router;
